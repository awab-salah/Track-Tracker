     1→     1→/**
     2→     2→ * ZainCash Payment Gateway routes for the Express API server.
     3→     3→ *
     4→     4→ * Endpoints:
     5→     5→ *   POST /api/zaincash/create  — Create a payment transaction
     6→     6→ *   POST /api/zaincash/callback — Server-to-server callback from ZainCash
     7→     7→ *   GET  /api/zaincash/verify  — Verify payment status (client polling)
     8→     8→ *
     9→     9→ * Uses ZainCash v1 API (JWT-based authentication).
    10→    10→ *
    11→    11→ * IMPORTANT: The v2 OAuth2 API (/api/v2/oauth2/token and
    12→    12→ * /api/v2/payment-gateway/transaction/init) is behind Cloudflare WAF
    13→    13→ * and returns 403 Forbidden for server-to-server requests from
    14→    14→ * serverless environments. All production ZainCash integrations
    15→    15→ * (Laravel package, Flutter packages) use the v1 JWT flow.
    16→    16→ *
    17→    17→ * v1 Flow:
    18→    18→ *   1. Create JWT payload { amount, serviceType, msisdn, orderId, redirectUrl, iat, exp }
    19→    19→ *   2. Sign with HMAC-SHA256 using the merchant secret key
    20→    20→ *   3. POST to {baseUrl}/transaction/init with { lang, merchantId, token }
    21→    21→ *   4. Response: { id, rUrl } — redirect user to rUrl + id
    22→    22→ *
    23→    23→ * Inquiry (for verify/callback):
    24→    24→ *   1. Create JWT payload { id, msisdn, iat, exp }
    25→    25→ *   2. POST to {baseUrl}/transaction/get with { merchantId, token }
    26→    26→ *
    27→    27→ * Sandbox defaults: official test credentials from the ZainCash Laravel package
    28→    28→ * (https://github.com/waadmawlood/zaincash).
    29→    29→ */

    66→    66→function base64UrlEncode(data: string): string {
    67→    67→  return Buffer.from(data, 'utf-8')
    68→    68→    .toString('base64')
    69→    69→    .replace(/=/g, '')
    70→    70→    .replace(/\+/g, '-')
    71→    71→    .replace(/\//g, '_');
    72→    72→}
    73→    73→
    74→    74→/**
    75→    75→ * Create a JWT token signed with HMAC-SHA256.
    76→    76→ * Used for both transaction init and inquiry requests.
    77→    77→ */
    78→    78→function createJWT(payload: Record<string, unknown>, secret: string): string {
    79→    79→  const header = { alg: 'HS256', typ: 'JWT' };
    80→    80→  const headerB64 = base64UrlEncode(JSON.stringify(header));
    81→    81→  const payloadB64 = base64UrlEncode(JSON.stringify(payload));
    82→    82→  const signature = crypto
    83→    83→    .createHmac('sha256', secret)
    84→    84→    .update(`${headerB64}.${payloadB64}`)
    85→    85→    .digest('base64')
    86→    86→    .replace(/=/g, '')
    87→    87→    .replace(/\+/g, '-')
    88→    88→    .replace(/\//g, '_');
    89→    89→  return `${headerB64}.${payloadB64}.${signature}`;
    90→    90→}
    91→    91→
    92→    92→/**
    93→    93→ * Decode a ZainCash callback JWT token (v1).
    94→    94→ * Returns the payload if valid, throws if invalid.
    95→    95→ */
    96→    96→function decodeJwt(token: string, secret: string): Record<string, unknown> {
    97→    97→  const parts = token.split('.');
    98→    98→  if (parts.length !== 3) throw new Error('Invalid JWT format');
    99→    99→  const payloadB64 = parts[1].replace(/-/g, '+').replace(/_/g, '/');
   100→   100→  const payloadJson = Buffer.from(payloadB64, 'base64').toString('utf-8');
   101→   101→  const signature = crypto
   102→   102→    .createHmac('sha256', secret)
   103→   103→    .update(`${parts[0]}.${parts[1]}`)
   104→   104→    .digest('base64')
   105→   105→    .replace(/=/g, '')
   106→   106→    .replace(/\+/g, '-')
   107→   107→    .replace(/\//g, '_');
   108→   108→  if (signature !== parts[2]) throw new Error('JWT signature verification failed');
   109→   109→  return JSON.parse(payloadJson);
   110→   110→}

   120→   120→// ── Inquiry helper (v1 API) ──────────────────────────────────────────────────
   121→   121→
   122→   122→async function inquireTransaction(transactionId: string): Promise<Record<string, unknown>> {
   123→   123→  const config = getConfig();
   124→   124→  const now = Math.floor(Date.now() / 1000);
   125→   125→  const jwtPayload = {
   126→   126→    id: transactionId,
   127→   127→    msisdn: config.msisdn,
   128→   128→    iat: now,
   129→   129→    exp: now + 60 * 60 * 4,
   130→   130→  };
   131→   131→  const token = createJWT(jwtPayload, config.secret);
   132→   132→
   133→   133→  const inquiryUrl = `${config.baseUrl}/transaction/get`;
   134→   134→  console.log('[ZainCash] Inquiry request to:', inquiryUrl, 'for transaction:', transactionId);
   135→   135→
   136→   136→  // ZainCash v1 API requires application/x-www-form-urlencoded, NOT JSON
   137→   137→  const inquiryParams = new URLSearchParams();
   138→   138→  inquiryParams.append('merchantId', config.merchantId);
   139→   139→  inquiryParams.append('token', token);
   140→   140→
   141→   141→  const response = await fetch(inquiryUrl, {
   142→   142→    method: 'POST',
   143→   143→    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   144→   144→    body: inquiryParams.toString(),
   145→   145→  });
   146→   146→
   147→   147→  const responseText = await response.text();
   148→   148→  console.log('[ZainCash] Inquiry response status:', response.status);
   149→   149→  console.log('[ZainCash] Inquiry response body:', responseText);
   150→   150→
   151→   151→  if (!response.ok) {
   152→   152→    throw new Error(`ZainCash inquiry failed: ${response.status} — ${responseText.slice(0, 500)}`);
   153→   153→  }
   154→   154→
   155→   155→  let data: Record<string, unknown>;
   156→   156→  try {
   157→   157→    data = JSON.parse(responseText);
   158→   158→  } catch {
   159→   159→    throw new Error(`ZainCash inquiry returned invalid JSON: ${responseText.slice(0, 200)}`);
   160→   160→  }
   161→   161→
   162→   162→  if (data.err) {
   163→   163→    throw new Error(`ZainCash inquiry error: ${JSON.stringify(data.err)}`);
   164→   164→  }
   165→   165→
   166→   166→  return data;
   167→   167→}

   173→   173→/**
   174→   174→ * POST /api/zaincash/create
   175→   175→ * Create a ZainCash payment transaction using v1 JWT API.
   176→   176→ */
   177→   177→router.post('/zaincash/create', async (req: Request, res: Response) => {
   178→   178→  try {
   179→   179→    const { planId, amount, companyId } = req.body as {
   180→   180→      planId?: string;
   181→   181→      amount?: number;
   182→   182→      companyId?: string;
   183→   183→    };
   184→   184→
   185→   185→    if (!planId || !amount || !companyId) {
   186→   186→      return res.status(400).json({ error: 'Missing required fields: planId, amount, companyId' });
   187→   187→    }
   188→   188→
   189→   189→    if (amount <= 0) {
   190→   190→      return res.status(400).json({ error: 'Amount must be positive' });
   191→   191→    }
   192→   192→
   193→   193→    const config = getConfig();
   194→   194→
   195→   195→    // v1 API requires msisdn + merchantId + secret
   196→   196→    if (!config.msisdn || !config.merchantId || !config.secret) {
   197→   197→      return res.status(503).json({
   198→   198→        error: 'ZainCash payment gateway not configured',
   199→   199→        details: 'Missing required v1 credentials: msisdn, merchantId, or secret key',
   200→   200→      });
   201→   201→    }
   202→   202→
   203→   203→    // Generate unique order ID
   204→   204→    const orderId = `tt-${planId}-${companyId}-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   205→   205→
   206→   206→    // Build redirectUrl for the JWT payload.
   207→   207→    // In ZainCash v1, redirectUrl serves as BOTH user redirect and callback:
   208→   208→    //   - After payment, ZainCash redirects the user's browser to redirectUrl?token=XXXXX
   209→   209→    //   - The token JWT contains the payment result (status, orderId, id)
   210→   210→    // We append orderId/planId/companyId as query params so the callback endpoint
   211→   211→    // can use them as fallback if the token doesn't contain enough info.
   212→   212→    const redirectUrl = `${config.redirectUrl}?orderId=${orderId}&planId=${planId}&companyId=${companyId}`;
   213→   213→
   214→   214→    // ── Step 1: Create JWT token ──────────────────────────────────────────
   215→   215→    const now = Math.floor(Date.now() / 1000);
   216→   216→    const jwtPayload = {
   217→   217→      amount,
   218→   218→      serviceType: 'subscription',
   219→   219→      msisdn: config.msisdn,
   220→   220→      orderId,
   221→   221→      redirectUrl,  // ZainCash v1 callback: browser redirect to this URL + ?token=XXXXX
   222→   222→      iat: now,
   223→   223→      exp: now + 60 * 60 * 4, // 4 hours expiry
   224→   224→    };
   225→   225→
   226→   226→    const token = createJWT(jwtPayload, config.secret);
   227→   227→    console.log('[ZainCash] JWT token created for order:', orderId);
   228→   228→
   229→   229→    // ── Step 2: Initiate transaction via v1 API ───────────────────────────
   230→   230→    // ZainCash v1 API requires application/x-www-form-urlencoded, NOT JSON
   231→   231→    const initUrl = `${config.baseUrl}/transaction/init`;
   232→   232→    const initParams = new URLSearchParams();
   233→   233→    initParams.append('token', token);
   234→   234→    initParams.append('merchantId', config.merchantId);
   235→   235→    initParams.append('lang', config.lang);
   236→   236→
   237→   237→    console.log('[ZainCash] Initiating transaction at:', initUrl);
   238→   238→    console.log('[ZainCash] Content-Type: application/x-www-form-urlencoded');
   239→   239→
   240→   240→    const initResponse = await fetch(initUrl, {
   241→   241→      method: 'POST',
   242→   242→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   243→   243→      body: initParams.toString(),
   244→   244→    });
   245→   245→
   246→   246→    const initText = await initResponse.text();
   247→   247→    console.log('[ZainCash] Init response status:', initResponse.status);
   248→   248→    console.log('[ZainCash] Init response body:', initText);
   249→   249→
   250→   250→    if (!initResponse.ok) {
   251→   251→      console.error('[ZainCash] Init failed:', initResponse.status, initText);
   252→   252→      return res.status(502).json({
   253→   253→        error: 'ZainCash transaction init failed',
   254→   254→        step: 'transaction_init',
   255→   255→        zaincashStatus: initResponse.status,
   256→   256→        zaincashResponse: initText.slice(0, 1000),
   257→   257→      });
   258→   258→    }
   259→   259→
   260→   260→    let initData: { id?: string; rUrl?: string; err?: { msg?: string } };
   261→   261→    try {
   262→   262→      initData = JSON.parse(initText);
   263→   263→    } catch {
   264→   264→      console.error('[ZainCash] Invalid JSON response:', initText);
   265→   265→      return res.status(502).json({
   266→   266→        error: 'Invalid JSON from ZainCash',
   267→   267→        step: 'transaction_init',
   268→   268→        zaincashResponse: initText.slice(0, 500),
   269→   269→      });
   270→   270→    }
   271→   271→
   272→   272→    // Check for error in response body (ZainCash returns { err: { msg } })
   273→   273→    if (initData.err) {
   274→   274→      console.error('[ZainCash] ZainCash error:', initData.err);
   275→   275→      return res.status(502).json({
   276→   276→        error: 'ZainCash rejected the transaction',
   277→   277→        step: 'transaction_init',
   278→   278→        zaincashError: initData.err,
   279→   279→      });
   280→   280→    }
   281→   281→
   282→   282→    if (!initData.id) {
   283→   283→      console.error('[ZainCash] No transaction ID in response:', initData);
   284→   284→      return res.status(502).json({
   285→   285→        error: 'No transaction ID from ZainCash',
   286→   286→        step: 'transaction_init',
   287→   287→        zaincashResponse: initData,
   288→   288→      });
   289→   289→    }
   290→   290→
   291→   291→    // v1 API returns { id, rUrl } where rUrl is the base redirect URL
   292→   292→    // Full redirect URL = rUrl + id
   293→   293→    const redirectUrlFull = initData.rUrl
   294→   294→      ? `${initData.rUrl}${initData.id}`
   295→   295→      : `${config.baseUrl}/transaction/pay?id=${initData.id}`;
   296→   296→
   297→   297→    console.log('[ZainCash] Transaction created:', initData.id, 'for order:', orderId);
   298→   298→    console.log('[ZainCash] Redirect URL:', redirectUrlFull);
   299→   299→
   300→   300→    // Store payment record in Supabase (fire-and-forget)
   301→   301→    try {
   302→   302→      const { url: supabaseUrl, key: supabaseKey } = getSupabaseCreds();
   303→   303→      if (supabaseUrl && supabaseKey) {
   304→   304→        await fetch(`${supabaseUrl}/rest/v1/payment_records`, {
   305→   305→          method: 'POST',
   306→   306→          headers: {
   307→   307→            'Content-Type': 'application/json',
   308→   308→            apikey: supabaseKey,
   309→   309→            Authorization: `Bearer ${supabaseKey}`,
   310→   310→            Prefer: 'return=minimal',
   311→   311→          },
   312→   312→          body: JSON.stringify({
   313→   313→            id: initData.id,
   314→   314→            order_id: orderId,
   315→   315→            company_id: companyId,
   316→   316→            plan_id: planId,
   317→   317→            amount,
   318→   318→            status: 'pending',
   319→   319→            created_at: new Date().toISOString(),
   320→   320→          }),
   321→   321→        });
   322→   322→      }
   323→   323→    } catch (dbErr) {
   324→   324→      console.warn('[ZainCash] Failed to store payment record:', dbErr);
   325→   325→    }
   326→   326→
   327→   327→    return res.status(200).json({
   328→   328→      transactionId: initData.id,
   329→   329→      redirectUrl: redirectUrlFull,
   330→   330→      orderId,
   331→   331→    });
   332→   332→
   333→   333→  } catch (err) {
   334→   334→    const message = err instanceof Error ? err.message : String(err);
   335→   335→    const stack = err instanceof Error ? err.stack : undefined;
   336→   336→    console.error('[ZainCash] Create payment error:', message);
   337→   337→    console.error('[ZainCash] Stack:', stack);
   338→   338→    // Return the REAL error so the caller knows exactly which step failed
   339→   339→    return res.status(500).json({
   340→   340→      error: message,
   341→   341→      step: 'create_payment',
   342→   342→    });
   343→   343→  }
   344→   344→});

   383→   383→// ── Core callback processing (shared by GET and POST) ─────────────────────────
   384→   384→
   385→   385→async function processCallback(params: {
   386→   386→  transactionId: string;
   387→   387→  orderId?: string;
   388→   388→  planId?: string;
   389→   389→  companyId?: string;
   390→   390→}): Promise<{ success: boolean; transactionId: string; status: string; message: string }> {
   391→   391→  const { transactionId } = params;
   392→   392→
   393→   393→  if (!transactionId) {
   394→   394→    return { success: false, transactionId: '', status: 'invalid', message: 'Missing transaction ID' };
   395→   395→  }
   396→   396→
   397→   397→  let companyId = params.companyId ?? '';
   398→   398→
   399→   399→  // ── Idempotency check: if payment already completed, skip ──────────────
   400→   400→  try {
   401→   401→    const { url: supabaseUrl, key: supabaseKey } = getSupabaseCreds();
   402→   402→    if (supabaseUrl && supabaseKey) {
   403→   403→      const recordRes = await fetch(
   404→   404→        `${supabaseUrl}/rest/v1/payment_records?id=eq.${transactionId}&select=status,company_id`,
   405→   405→        { method: 'GET', headers: { 'Content-Type': 'application/json', apikey: supabaseKey, Authorization: `Bearer ${supabaseKey}` } },
   406→   406→      );
   407→   407→      if (recordRes.ok) {
   408→   408→        const rows = await recordRes.json() as Array<{ status: string; company_id: string }>;
   409→   409→        if (rows.length > 0) {
   410→   410→          if (rows[0].status === 'completed') {
   411→   411→            console.log('[ZainCash] Callback: payment already completed (idempotent):', transactionId);
   412→   412→            return { success: true, transactionId, status: 'completed', message: 'Already processed' };
   413→   413→          }
   414→   414→          // Use company_id from payment_records (most reliable)
   415→   415→          if (!companyId && rows[0].company_id) {
   416→   416→            companyId = rows[0].company_id;
   417→   417→          }
   418→   418→        }
   419→   419→      }
   420→   420→    }
   421→   421→  } catch { /* non-fatal — continue without idempotency check */ }
   422→   422→
   423→   423→  // Always verify via inquiry — NEVER trust the callback payload alone
   424→   424→  console.log('[ZainCash] Verifying transaction via inquiry:', transactionId);
   425→   425→  const details = await inquireTransaction(transactionId) as {
   426→   426→    status: string;
   427→   427→    orderId: string;
   428→   428→  };
   429→   429→
   430→   430→  // Use companyId from payment_records (already set above) or fallback
   431→   431→  if (!companyId) {
