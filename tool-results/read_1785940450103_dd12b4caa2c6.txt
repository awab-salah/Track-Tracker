   104→   104→function generateJoinCode(): string {
   105→   105→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
   106→   106→  return Array.from({ length: 6 }, () => chars[Math.floor(Math.random() * chars.length)]).join('');
   107→   107→}

   109→   109→function generateId(): string {
   110→   110→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
   111→   111→    return crypto.randomUUID();
   112→   112→  }
   113→   113→  return `local-${Date.now().toString(36)}-${Math.floor(Math.random() * 1e4)}`;
   114→   114→}

   124→   124→// Local-only preference — Notification permission itself is a per-browser/device
   125→   125→// concept (not a company-level setting), so it is intentionally not persisted
   126→   126→// to Supabase. Kept in localStorage like the dark-mode toggle.
   127→   127→const NOTIFICATIONS_STORAGE_KEY = 'tt_notifications_enabled';
   128→   128→
   129→   129→/**
   130→   130→ * Build the localStorage key for the "cargo edited today" latch.
   131→   131→ *
   132→   132→ * Keyed by `(driverId, today)` so the flag:
   133→   133→ *   - auto-resets at midnight (today changes → new key → flag starts false),
   134→   134→ *   - is per-driver (multiple drivers on the same browser don't collide),
   135→   135→ *   - survives page refresh (same key → same value).
   136→   136→ *
   137→   137→ * This latch is the mechanism that flips Day-2's cargo title from
   138→   138→ * "الحمولة المتبقية من اليوم السابق" to "الحمولة الحالية" after the driver's
   139→   139→ * FIRST cargo mutation of the day. The carry-over algorithm itself is NOT
   140→   140→ * modified — only the title-resolution layer consumes this flag.
   141→   141→ */
   142→   142→function cargoEditedTodayStorageKey(driverId: string, today: string): string {
   143→   143→  return `tt_cargo_edited_${driverId}_${today}`;
   144→   144→}
   145→   145→
   146→   146→/** Read the latch from localStorage. Returns false if not set or unavailable. */
   147→   147→function readCargoEditedToday(driverId: string, today: string): boolean {
   148→   148→  if (typeof window === 'undefined') return false;
   149→   149→  try {
   150→   150→    return localStorage.getItem(cargoEditedTodayStorageKey(driverId, today)) === '1';
   151→   151→  } catch {
   152→   152→    return false;
   153→   153→  }
   154→   154→}
   155→   155→
   156→   156→/** Write the latch to localStorage. Silently no-op if storage is unavailable. */
   157→   157→function writeCargoEditedToday(driverId: string, today: string, value: boolean): void {
   158→   158→  if (typeof window === 'undefined') return;
   159→   159→  try {
   160→   160→    if (value) {
   161→   161→      localStorage.setItem(cargoEditedTodayStorageKey(driverId, today), '1');
   162→   162→    } else {
   163→   163→      localStorage.removeItem(cargoEditedTodayStorageKey(driverId, today));
   164→   164→    }
   165→   165→  } catch {
   166→   166→    /* ignore storage failures — flag is also kept in React state */
   167→   167→  }
   168→   168→}

   171→   171→  if (typeof window === 'undefined' || !('Notification' in window)) return 'unsupported';

   173→   173→}

   175→   175→function dedupeLoads(loads: CargoItem[]): CargoItem[] {
   176→   176→  const byKey = new Map<string, CargoItem>();
   177→   177→  for (const load of loads) {
   178→   178→    const key = `${load.driverId}::${load.productName.trim().toLowerCase()}`;
   179→   179→    const existing = byKey.get(key);
   180→   180→    if (existing) {
   181→   181→      existing.quantity += load.quantity;
   182→   182→      existing.unitPrice = load.unitPrice;
   183→   183→    } else {
   184→   184→      byKey.set(key, { ...load });
   185→   185→    }
   186→   186→  }
   187→   187→  return Array.from(byKey.values());
   188→   188→}

   214→   214→
   215→   215→  // ── Company subscription active (driver context) ──────────────────────────
   216→   216→  // For company owners: comes from `company.subscriptionActive` (set during
   217→   217→  // bootstrap from AuthContext's companyProfile). For drivers: fetched from
   218→   218→  // the parent company's `subscription_active` column via Supabase query
   219→   219→  // during the driver bootstrap. Defaults to false so the gate blocks until

   224→   224→  //

   226→   226→  // (add / remove / sell / change quantity / edit price / promote snapshot to
   227→   227→  // live) since midnight. Used by `useCargoHistory` to flip the Day-2 carry-over
   228→   228→  // title from "الحمولة المتبقية من اليوم السابق" to "الحمولة الحالية" on
   229→   229→  // the first mutation of the day. See the type-doc on `cargoEditedToday`
   230→   230→  // above for the full spec.
   231→   231→  //
   232→   232→  // `today` is recomputed on every render; if the calendar day rolls over
   233→   233→  // while the dashboard is open, the latch re-reads from localStorage under
   234→   234→  // the new key (which starts unset), effectively resetting the flag at

   254→   254→    writeCargoEditedToday(currentDriverId, today, true);

   263→   263→    NotificationPermission | 'unsupported'
   264→   264→  >(getNotificationPermission);
   265→   265→
   266→   266→  // Kept in a ref (not a dependency of the subscription effect below) so that
   267→   267→  // routine driver-list refreshes don't tear down and re-create the realtime
   268→   268→  // channel — only role/company/toggle changes should do that.
   269→   269→  const driversRef = useRef<Driver[]>(drivers);
   270→   270→  useEffect(() => {
   271→   271→    driversRef.current = drivers;
   272→   272→  }, [drivers]);

   274→   274→  // ── Dark mode ──────────────────────────────────────────────────────────────

   279→   279→
   280→   280→  // ── Bootstrap company data ─────────────────────────────────────────────────
   281→   281→  // Runs when a company owner signs in. Uses a cancellation flag so that
   282→   282→  // if auth changes before the fetch completes, stale results are discarded.
   283→   283→  useEffect(() => {
   284→   284→    if (authLoading) return;
   285→   285→    if (role !== 'company' || !authCompanyId || !authCompany) return;
   286→   286→    if (!isSupabaseConfigured) {
   287→   287→      setCompany(authCompany);
   288→   288→      return;
   289→   289→    }
   290→   290→
   291→   291→    let cancelled = false;
   292→   292→    setCompany(authCompany);
   293→   293→
   294→   294→    void (async () => {
   295→   295→      const remoteDrivers = await fetchDrivers(authCompanyId, authCompany.name);

   300→   300→        fetchLoads(driverIds),
   301→   301→        fetchSales(driverIds),
   302→   302→      ]);
   303→   303→      if (cancelled) return;
   304→   304→

   307→   307→      setSales(remoteSales);
   308→   308→      setCurrentDriverId(null);

   312→   312→      // Reads current DB loads (no mutation has happened today yet at bootstrap).

   314→   314→        remoteDrivers.map((d) => finalizeYesterdayIfNeeded(d.id))
   315→   315→      );
   316→   316→    })();
   317→   317→
   318→   318→    return () => { cancelled = true; };
   319→   319→  // authCompany?.name is a stable string proxy for the company object reference

   363→   363→        }
   364→   364→      }
   365→   365→
   366→   366→      const [remoteLoads, remoteSales] = await Promise.all([
   367→   367→        fetchLoads([authDriverId]),
   368→   368→        fetchSales([authDriverId]),
   369→   369→      ]);

   371→   371→      setLoads(dedupeLoads(remoteLoads));
   372→   372→      setSales(remoteSales);
   373→   373→
   374→   374→      // Finalize yesterday's snapshot for this driver. Fire-and-forget —
   375→   375→      // idempotent. Reads current DB loads (no mutation today yet).
   376→   376→      void finalizeYesterdayIfNeeded(authDriverId);
   377→   377→    })();
   378→   378→
   379→   379→    return () => { cancelled = true; };
   380→   380→  // eslint-disable-next-line react-hooks/exhaustive-deps
   381→   381→  }, [role, authDriverId, authLoading]);

   387→   387→      setDrivers([]);
   388→   388→      setLoads([]);
   389→   389→      setSales([]);
   390→   390→      setCurrentDriverId(null);
   391→   391→      setDriverCompanySubscriptionActive(false);
   392→   392→    }
   393→   393→  }, [role, authLoading]);

   420→   420→      void updateCompany(authCompanyId, { joinCode: code });
   421→   421→    }
   422→   422→  };
   423→   423→
   424→   424→  const regenerateJoinCode = () => {
   425→   425→    const newCode = generateJoinCode();
   426→   426→    setCompany((prev) => ({ ...prev, joinCode: newCode }));
   427→   427→    if (authCompanyId) {
   428→   428→      void updateCompany(authCompanyId, { joinCode: newCode });
   429→   429→    }
   430→   430→  };

   432→   432→  // ── Subscription activation ──────────────────────────────────────────────
   433→   433→  //
   434→   434→  // Uses the Supabase `activate_subscription` security-definer RPC to validate
   435→   435→  // the activation code and set `subscription_active = true` on the company row.
   436→   436→  // Activation persists ONLY in Supabase — no localStorage, no fallback.
   437→   437→  //
   438→   438→  // The only valid activation code for now is "track1".
   439→   439→  // Later Stripe will replace this — the architecture is ready.
   440→   440→  //
   441→   441→  // Activation strategy (ordered by preference):
   442→   442→  //   1. Call the `activate_subscription` security-definer RPC. This validates
   443→   443→  //      the code server-side and persists to the DB in one atomic call.
   444→   444→  //   2. If the RPC doesn't exist, validate the code locally and do a direct
   445→   445→  //      PostgREST UPDATE on the company row.
   446→   446→  //      This works because company owners have UPDATE RLS on their own row.
   447→   447→  //   3. If both strategies fail, return false (activation not persisted).
   448→   448→  //      The user sees an error and can retry later.
   449→   449→
   450→   450→  const activateSubscription = async (code: string): Promise<boolean> => {
   451→   451→    // If the company is already active, return true immediately.
   452→   452→    // The UI layer checks companySubscriptionActive before calling this
   453→   453→    // function and shows an "already activated" message, but this guard
   454→   454→    // ensures that even if the UI bypass is missed, no duplicate activation
   455→   455→    // attempt reaches the database.
   456→   456→    if (company.subscriptionActive) return true;
   457→   457→
   458→   458→    const normalizedCode = code.trim().toLowerCase();
   459→   459→    const isValid = normalizedCode === 'track1';
   460→   460→
   461→   461→    if (!isValid) return false;
   462→   462→
   463→   463→    if (!isSupabaseConfigured) {
   464→   464→      // Offline mode: accept "track1" and update local state only.
   465→   465→      // No localStorage — state lives only in React (lost on refresh,
   466→   466→      // which is correct since the DB is unreachable in offline mode).
   467→   467→      setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   468→   468→      return true;
   469→   469→    }
   470→   470→
   471→   471→    // ── Strategy 1: try the RPC ──
   472→   472→    try {
   473→   473→      const { data, error } = await supabase.rpc('activate_subscription', {
   474→   474→        p_activation_code: code.trim(),
   475→   475→      });
   476→   476→
   477→   477→      if (!error) {
   478→   478→        const success = data as boolean;
   479→   479→        if (success) {
   480→   480→          setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   481→   481→        }
   482→   482→        return success;
   483→   483→      }
   484→   484→
   485→   485→      // RPC doesn't exist or failed — proceed to fallback strategy.
   486→   486→      if (error.message.includes('Could not find the function') || error.code === 'PGRST202') {
   487→   487→        console.warn('[AppContext] activate_subscription RPC not found — falling back to direct UPDATE.');
   488→   488→      } else {
   489→   489→        console.warn('[AppContext] activate_subscription RPC error:', error.message, '— falling back to direct UPDATE.');
   490→   490→      }
   491→   491→    } catch (err) {
   492→   492→      console.warn('[AppContext] activate_subscription RPC error:', err, '— falling back to direct UPDATE.');
   493→   493→    }
   494→   494→
   495→   495→    // ── Strategy 2: direct PostgREST UPDATE ──
   496→   496→    // Company owners have `companies_owner` RLS policy which allows
   497→   497→    // UPDATE on their own row. This persists activation to the DB.
   498→   498→    const companyId = authCompanyId;
   499→   499→    if (companyId) {
   500→   500→      try {
   501→   501→        const { error: updateError } = await supabase
   502→   502→          .from('companies')
   503→   503→          .update({ subscription_active: true })
   504→   504→          .eq('id', companyId);
   505→   505→
   506→   506→        if (!updateError) {
   507→   507→          setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   508→   508→          return true;
   509→   509→        }
   510→   510→
   511→   511→        console.warn('[AppContext] direct UPDATE failed:', updateError.message);
   512→   512→      } catch (err) {
   513→   513→        console.warn('[AppContext] direct UPDATE error:', err);
   514→   514→      }
   515→   515→    }
   516→   516→
   517→   517→    // ── Strategy 3: both strategies failed — activation not persisted ──
   518→   518→    // Return false so the user sees an error and can retry.
   519→   519→    return false;
   520→   520→  };
   521→   521→
   522→   522→  // ── Auth actions ──────────────────────────────────────────────────────────
   523→   523→
   524→   524→  const logout = () => {
   525→   525→    void signOut().then(() => setLocation('/'));
   526→   526→  };
   527→   527→
   528→   528→  const logoutDriver = () => {
   529→   529→    void signOut().then(() => setLocation('/'));
   530→   530→  };
   531→   531→
   532→   532→  // ── Driver profile ────────────────────────────────────────────────────────
   533→   533→
   534→   534→  const updateDriverProfile = (
   535→   535→    data: Partial<Pick<Driver, 'name' | 'email' | 'vehicleNumber' | 'profilePictureUrl'>>
   536→   536→  ) => {
   537→   537→    if (!currentDriverId) return;
   538→   538→    setDrivers((prev) =>
   539→   539→      prev.map((d) => (d.id === currentDriverId ? { ...d, ...data } : d))
   540→   540→    );
   541→   541→    void updateDriver(currentDriverId, data);
   542→   542→  };
   543→   543→
   544→   544→  // ── Load management ───────────────────────────────────────────────────────
   545→   545→
   546→   546→  const upsertLoad = (input: {
   547→   547→    id?: string;
   548→   548→    productName: string;
   549→   549→    quantity: number;
   550→   550→    unitPrice: number;
   551→   551→  }) => {
   552→   552→    if (!currentDriverId) return;
   553→   553→    const productName = input.productName.trim();
   554→   554→
   555→   555→    // Capture pre-mutation cargo so the finalize call below freezes yesterday's
   556→   556→    // EOD state, not the post-edit state. UI updates synchronously via setLoads.
   557→   557→    const preMutationCargo = loads.filter((l) => l.driverId === currentDriverId);
   558→   558→
   559→   559→    setLoads((prev) => {
   560→   560→      const existingIdx = prev.findIndex(
   561→   561→        (l) =>
   562→   562→          l.driverId === currentDriverId &&
   563→   563→          l.productName.trim().toLowerCase() === productName.toLowerCase()
   564→   564→      );
   565→   565→
   566→   566→      if (existingIdx !== -1) {
   567→   567→        const updated = [...prev];
   568→   568→        const existing = { ...updated[existingIdx], quantity: input.quantity, unitPrice: input.unitPrice };
   569→   569→        updated[existingIdx] = existing;
   570→   570→
   571→   571→        void dbUpsertLoad({
   572→   572→          id: existing.id,
   573→   573→          driverId: currentDriverId,
   574→   574→          productName: existing.productName,
   575→   575→          quantity: existing.quantity,
   576→   576→          unitPrice: existing.unitPrice,
   577→   577→          isNew: false,
   578→   578→        });
   579→   579→
   580→   580→        return updated;
   581→   581→      }
   582→   582→
   583→   583→      const newId = input.id ?? generateId();
   584→   584→      const newItem: CargoItem = {
   585→   585→        id: newId,
   586→   586→        driverId: currentDriverId,
   587→   587→        productName,
   588→   588→        quantity: input.quantity,
   589→   589→        unitPrice: input.unitPrice,
   590→   590→      };
   591→   591→
   592→   592→      void dbUpsertLoad({
   593→   593→        id: newId,
   594→   594→        driverId: currentDriverId,
   595→   595→        productName,
   596→   596→        quantity: input.quantity,
   597→   597→        unitPrice: input.unitPrice,
   598→   598→        isNew: true,
   599→   599→      });
   600→   600→
   601→   601→      return [...prev, newItem];
   602→   602→    });
   603→   603→
   604→   604→    // Finalize yesterday's snapshot AFTER local state update, using captured
   605→   605→    // pre-mutation state. Fire-and-forget — idempotent.
   606→   606→    void finalizeYesterdayIfNeeded(currentDriverId, preMutationCargo);
   607→   607→
   608→   608→    // Latch the "cargo edited today" flag so the Day-2 carry-over title
   609→   609→    // immediately flips from "الحمولة المتبقية من اليوم السابق" to
   610→   610→    // "الحمولة الحالية". See the type-doc on `cargoEditedToday`.
   611→   611→    markCargoEditedToday();
   612→   612→  };
   613→   613→
   614→   614→  const removeLoad = (id: string) => {
   615→   615→    if (!currentDriverId) return;
   616→   616→    // Capture pre-mutation cargo for the finalize call below.
   617→   617→    const preMutationCargo = loads.filter((l) => l.driverId === currentDriverId);
   618→   618→    setLoads((prev) => prev.filter((l) => l.id !== id));
   619→   619→    void dbRemoveLoad(id);
   620→   620→    void finalizeYesterdayIfNeeded(currentDriverId, preMutationCargo);
   621→   621→    // Latch the "cargo edited today" flag — removing a product is an edit.
   622→   622→    markCargoEditedToday();
   623→   623→  };
   624→   624→
   625→   625→  // ── Historical snapshot → live cargo promotion ─────────────────────────────
   626→   626→  //
   627→   627→  // Per spec: when the driver edits any product from a past day's "Remaining
   628→   628→  // Cargo From This Day", that snapshot immediately becomes the live Current
   629→   629→  // Cargo. We:
   630→   630→  //   1. Fetch the snapshot for the given date.
   631→   631→  //   2. Filter out qty-0 items (already hidden in the UI, but defensive).
   632→   632→  //   3. Assign fresh UUIDs to each item (new live rows, not snapshot rows).
   633→   633→  //   4. Optimistically replace the driver's loads in local state.
   634→   634→  //   5. Atomically replace the driver's loads in the DB.
   635→   635→  //   6. Return the new live CargoItem[] so the caller can find the matching
   636→   636→  //      item and open the Load tab with it prefilled.
   637→   637→  //
   638→   638→  // Note: this is destructive — the driver's previous live loads are lost.
   639→   639→  // This is the intended behavior per spec ("It immediately becomes Current
   640→   640→  // Cargo because it is now the active working inventory again").
   641→   641→  //
   642→   642→  // SAFETY: if the snapshot for `snapshotDate` does not exist (or is empty),
   643→   643→  // we ABORT the promotion and return the current live cargo unchanged.
   644→   644→  // Without this guard, a missing snapshot would cause `setLoads` to delete
   645→   645→  // ALL of the driver's current live cargo — a destructive no-op. See Bug 4
   646→   646→  // in the worklog.
   647→   647→  const promoteSnapshotToLive = async (
   648→   648→    driverId: string,
   649→   649→    snapshotDate: string
   650→   650→  ): Promise<CargoItem[]> => {
   651→   651→    const snapshotItems = await fetchDailySnapshots([driverId], snapshotDate);
   652→   652→    const itemsToPromote = snapshotItems.filter((item) => item.quantity > 0);
   653→   653→
   654→   654→    // SAFETY: no snapshot (or all-qty-0 snapshot) → abort promotion.
   655→   655→    // Return the driver's CURRENT live cargo unchanged so the caller can
   656→   656→    // still open the editor against the matching product. Without this
   657→   657→    // guard, a missing snapshot would cause `setLoads` to delete ALL of
   658→   658→    // the driver's current live cargo — a destructive no-op. See Bug 4
   659→   659→    // in the worklog.
   660→   660→    if (itemsToPromote.length === 0) {
   661→   661→      return loads.filter((l) => l.driverId === driverId);
   662→   662→    }
   663→   663→
   664→   664→    const newCargo: CargoItem[] = itemsToPromote.map((item) => ({
   665→   665→      id: generateId(),
   666→   666→      driverId,
   667→   667→      productName: item.productName,
   668→   668→      quantity: item.quantity,
   669→   669→      unitPrice: item.unitPrice,
   670→   670→    }));
   671→   671→
   672→   672→    // Optimistic local state update: remove old loads for this driver, add new.
   673→   673→    setLoads((prev) => [
   674→   674→      ...prev.filter((l) => l.driverId !== driverId),
   675→   675→      ...newCargo,
   676→   676→    ]);
   677→   677→
   678→   678→    // DB: atomically replace the driver's loads.
   679→   679→    await replaceDriverLoads(driverId, newCargo);
   680→   680→
   681→   681→    // Promoting a historical snapshot to live IS a cargo edit — latch the
   682→   682→    // "cargo edited today" flag so today's title becomes "الحمولة الحالية".
   683→   683→    // Only latches when the promotion targets the current driver (the
   684→   684→    // company-dashboard case where `driverId === currentDriverId`); the
   685→   685→    // company dashboard is read-only and never calls this with another
   686→   686→    // driver's id, but we guard anyway for safety.
   687→   687→    if (driverId === currentDriverId) {
   688→   688→      markCargoEditedToday();
   689→   689→    }
   690→   690→
   691→   691→    return newCargo;
   692→   692→  };
   693→   693→
   694→   694→  // ── Sales ─────────────────────────────────────────────────────────────────
   695→   695→
   696→   696→  const addSale = (items: SaleLineItem[], receiptImageUrl?: string | null) => {
   697→   697→    if (!currentDriverId) return;
   698→   698→
   699→   699→    // Capture pre-mutation cargo so the finalize call below freezes yesterday's
   700→   700→    // EOD state, not the post-sale (decremented) state. UI updates synchronously.
   701→   701→    const preMutationCargo = loads.filter((l) => l.driverId === currentDriverId);
   702→   702→
   703→   703→    setLoads((prev) =>
   704→   704→      prev.map((load) => {
   705→   705→        if (load.driverId !== currentDriverId) return load;
   706→   706→        const soldItem = items.find(
   707→   707→          (i) => i.productName.trim().toLowerCase() === load.productName.trim().toLowerCase()
   708→   708→        );
   709→   709→        if (!soldItem) return load;
   710→   710→        const newQty = Math.max(0, load.quantity - soldItem.quantity);
   711→   711→        void decrementLoad(currentDriverId, load.productName, soldItem.quantity);
   712→   712→        return { ...load, quantity: newQty };
   713→   713→      })
   714→   714→    );
   715→   715→
   716→   716→    // Finalize yesterday's snapshot AFTER local state update, using captured
   717→   717→    // pre-mutation state. Fire-and-forget — idempotent. Must come before the
   718→   718→    // createSale call so yesterday's EOD is frozen before today's first sale
   719→   719→    // is recorded, but it doesn't need to be awaited (the snapshot insert is
   720→   720→    // independent of the sale insert; both are async DB writes).
   721→   721→    void finalizeYesterdayIfNeeded(currentDriverId, preMutationCargo);
   722→   722→
   723→   723→    const totalPrice = items.reduce((sum, i) => sum + i.quantity * i.unitPrice, 0);
   724→   724→    // Baghdad local date (UTC+3, no DST). en-CA → 'YYYY-MM-DD'.
   725→   725→    // Previously used UTC which mis-stamped sales made between 00:00–03:00 Baghdad.
   726→   726→    const date = new Date().toLocaleDateString('en-CA', { timeZone: 'Asia/Baghdad' });
   727→   727→
   728→   728→    const newSale: SaleRecord = {
   729→   729→      id: generateId(),
   730→   730→      driverId: currentDriverId,
   731→   731→      date,
   732→   732→      items,
   733→   733→      totalPrice,
   734→   734→      receiptImageUrl: receiptImageUrl ?? null,
   735→   735→    };
   736→   736→    setSales((prev) => [newSale, ...prev]);
   737→   737→    void createSale(newSale.id, currentDriverId, date, items, totalPrice, receiptImageUrl ?? null);
   738→   738→
   739→   739→    // Per the revised midnight-logic spec, a sale IS a cargo mutation —
   740→   740→    // it decrements live quantities. Latch the "cargo edited today" flag
   741→   741→    // so the Day-2 carry-over title flips from "الحمولة المتبقية من اليوم
   742→   742→    // السابق" to "الحمولة الحالية" on the first sale of the day.
   743→   743→    markCargoEditedToday();
   744→   744→  };
   745→   745→
   746→   746→  // Only asks for browser permission when the owner explicitly enables the
   747→   747→  // toggle (never on load / never automatically).
   748→   748→  const enableNotifications = async () => {
   749→   749→    if (typeof window === 'undefined' || !('Notification' in window)) {
   750→   750→      setNotificationPermission('unsupported');
   751→   751→      return;
   752→   752→    }
   753→   753→
   754→   754→    let permission = Notification.permission;
   755→   755→    if (permission === 'default') {
   756→   756→      permission = await Notification.requestPermission();
   757→   757→    }
   758→   758→    setNotificationPermission(permission);
   759→   759→
   760→   760→    if (permission === 'granted') {
   761→   761→      setNotificationsEnabled(true);
   762→   762→      localStorage.setItem(NOTIFICATIONS_STORAGE_KEY, '1');
   763→   763→    } else {
   764→   764→      // Denied or dismissed — keep the toggle off.
   765→   765→      setNotificationsEnabled(false);
   766→   766→      localStorage.setItem(NOTIFICATIONS_STORAGE_KEY, '0');
   767→   767→    }
   768→   768→  };
   769→   769→
   770→   770→  const disableNotifications = () => {
   771→   771→    setNotificationsEnabled(false);
   772→   772→    localStorage.setItem(NOTIFICATIONS_STORAGE_KEY, '0');
   773→   773→  };
   774→   774→
   775→   775→  // ── Sale notifications — Supabase Realtime subscription ───────────────────
   776→   776→  // Fires a Web Notification the instant any driver in this company records a
   777→   777→  // new sale. Active only while: signed in as the company owner, the toggle is
   778→   778→  // on, and the browser permission is actually granted.
   779→   779→  useEffect(() => {
   780→   780→    if (role !== 'company' || !authCompanyId) return;
   781→   781→    if (!isSupabaseConfigured || !notificationsEnabled) return;
   782→   782→    if (getNotificationPermission() !== 'granted') return;
   783→   783→
   784→   784→    const channel = supabase
   785→   785→      .channel(`company-sales-notify-${authCompanyId}`)
   786→   786→      .on(
   787→   787→        'postgres_changes',
   788→   788→        { event: 'INSERT', schema: 'public', table: 'sales' },
   789→   789→        (payload) => {
   790→   790→          const row = payload.new as { driver_id: string; total_price: number };
   791→   791→          // sales has no company_id column — filter client-side against this
   792→   792→          // company's known driver ids (kept fresh via driversRef).
   793→   793→          const driver = driversRef.current.find((d) => d.id === row.driver_id);
   794→   794→          if (!driver) return;
   795→   795→
   796→   796→          try {
   797→   797→            new Notification('عملية بيع جديدة', {
   798→   798→              body: `${driver.name} سجّل عملية بيع بقيمة ${formatIQD(row.total_price)}`,
   799→   799→              icon: `${import.meta.env.BASE_URL}icons/icon-192.png`,
   800→   800→              tag: `sale-${row.driver_id}-${Date.now()}`,
   801→   801→            });
   802→   802→          } catch (err) {
   803→   803→            console.error('[AppContext] Failed to display sale notification:', err);
   804→   804→          }
   805→   805→        },
   806→   806→      )
   807→   807→      .subscribe();
   808→   808→
   809→   809→    return () => {
   810→   810→      void supabase.removeChannel(channel);
   811→   811→    };
   812→   812→  }, [role, authCompanyId, notificationsEnabled]);
   813→   813→
   814→   814→  const currentDriver = drivers.find((d) => d.id === currentDriverId) ?? null;
   815→   815→
   816→   816→  return (
   817→   817→    <AppContext.Provider
   818→   818→      value={{
   819→   819→        company,
   820→   820→        darkMode,
   821→   821→        toggleDarkMode,
   822→   822→        updateLogo,
   823→   823→        setCompanyProfile,
   824→   824→        setJoinCode,
   825→   825→        regenerateJoinCode,
   826→   826→        logout,
   827→   827→        activateSubscription,
   828→   828→        companySubscriptionActive: role === 'driver'
   829→   829→          ? driverCompanySubscriptionActive
   830→   830→          : company.subscriptionActive,
   831→   831→        drivers,
   832→   832→        loads,
   833→   833→        sales,
   834→   834→        currentDriverId,
   835→   835→        currentDriver,
   836→   836→        logoutDriver,
   837→   837→        updateDriverProfile,
   838→   838→        upsertLoad,
   839→   839→        removeLoad,
   840→   840→        promoteSnapshotToLive,
   841→   841→        addSale,
   842→   842→        cargoEditedToday,
   843→   843→        notificationsEnabled,
   844→   844→        notificationPermission,
   845→   845→        enableNotifications,
   846→   846→        disableNotifications,
   847→   847→      }}
   848→   848→    >
   849→   849→      {children}
   850→   850→    </AppContext.Provider>
   851→   851→  );
   852→   852→}
