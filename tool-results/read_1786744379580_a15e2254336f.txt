   126→function generateJoinCode(): string {
   127→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
   128→  return Array.from({ length: 6 }, () => chars[Math.floor(Math.random() * chars.length)]).join('');
   129→}

   131→function generateId(): string {
   132→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
   133→    return crypto.randomUUID();
   134→  }
   135→  return `local-${Date.now().toString(36)}-${Math.floor(Math.random() * 1e4)}`;
   136→}

   149→const NOTIFICATIONS_STORAGE_KEY = 'tt_notifications_enabled';
   150→
   151→/**
   152→ * Build the localStorage key for the "cargo edited today" latch.
   153→ *
   154→ * Keyed by `(driverId, today)` so the flag:
   155→ *   - auto-resets at midnight (today changes → new key → flag starts false),
   156→ *   - is per-driver (multiple drivers on the same browser don't collide),
   157→ *   - survives page refresh (same key → same value).
   158→ *
   159→ * This latch is the mechanism that flips Day-2's cargo title from
   160→ * "الحمولة المتبقية من اليوم السابق" to "الحمولة الحالية" after the driver's
   161→ * FIRST cargo mutation of the day. The carry-over algorithm itself is NOT
   162→ * modified — only the title-resolution layer consumes this flag.
   163→ */
   164→function cargoEditedTodayStorageKey(driverId: string, today: string): string {
   165→  return `tt_cargo_edited_${driverId}_${today}`;
   166→}
   167→
   168→/** Read the latch from localStorage. Returns false if not set or unavailable. */
   169→function readCargoEditedToday(driverId: string, today: string): boolean {
   170→  if (typeof window === 'undefined') return false;
   171→  try {
   172→    return localStorage.getItem(cargoEditedTodayStorageKey(driverId, today)) === '1';
   173→  } catch {
   174→    return false;
   175→  }
   176→}
   177→
   178→/** Write the latch to localStorage. Silently no-op if storage is unavailable. */
   179→function writeCargoEditedToday(driverId: string, today: string, value: boolean): void {
   180→  if (typeof window === 'undefined') return;
   181→  try {
   182→    if (value) {
   183→      localStorage.setItem(cargoEditedTodayStorageKey(driverId, today), '1');
   184→    } else {
   185→      localStorage.removeItem(cargoEditedTodayStorageKey(driverId, today));
   186→    }
   187→  } catch {
   188→    /* ignore storage failures — flag is also kept in React state */
   189→  }
   190→}

   197→function dedupeLoads(loads: CargoItem[]): CargoItem[] {
   198→  const byKey = new Map<string, CargoItem>();
   199→  for (const load of loads) {
   200→    const key = `${load.driverId}::${load.productName.trim().toLowerCase()}`;
   201→    const existing = byKey.get(key);
   202→    if (existing) {
   203→      existing.quantity += load.quantity;
   204→      existing.unitPrice = load.unitPrice;
   205→    } else {
   206→      byKey.set(key, { ...load });
   207→    }
   208→  }
   209→  return Array.from(byKey.values());
   210→}

   480→  // ── Subscription activation ──────────────────────────────────────────────
   481→  //
   482→  // Uses the Supabase `activate_subscription` security-definer RPC to validate
   483→  // the activation code and set `subscription_active = true` on the company row.
   484→  // Activation persists ONLY in Supabase — no localStorage, no fallback.
   485→  //
   486→  // The only valid activation code for now is "track1".
   487→  // Later Stripe will replace this — the architecture is ready.
   488→  //
   489→  // Activation strategy (ordered by preference):
   490→  //   1. Call the `activate_subscription` security-definer RPC. This validates
   491→  //      the code server-side and persists to the DB in one atomic call.
   492→  //   2. If the RPC doesn't exist, validate the code locally and do a direct
   493→  //      PostgREST UPDATE on the company row.
   494→  //      This works because company owners have UPDATE RLS on their own row.
   495→  //   3. If both strategies fail, return false (activation not persisted).
   496→  //      The user sees an error and can retry later.
   497→
   498→  const activateSubscription = async (code: string): Promise<boolean> => {
   499→    // If the company is already active, return true immediately.
   500→    // The UI layer checks companySubscriptionActive before calling this
   501→    // function and shows an "already activated" message, but this guard
   502→    // ensures that even if the UI bypass is missed, no duplicate activation
   503→    // attempt reaches the database.
   504→    if (company.subscriptionActive) return true;
   505→
   506→    const normalizedCode = code.trim().toLowerCase();
   507→    const isValid = normalizedCode === 'track1';
   508→
   509→    if (!isValid) return false;
   510→
   511→    if (!isSupabaseConfigured) {
   512→      // Offline mode: accept "track1" and update local state only.
   513→      // No localStorage — state lives only in React (lost on refresh,
   514→      // which is correct since the DB is unreachable in offline mode).
   515→      setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   516→      return true;
   517→    }
   518→
   519→    // ── Strategy 1: try the RPC ──
   520→    try {
   521→      const { data, error } = await supabase.rpc('activate_subscription', {
   522→        p_activation_code: code.trim(),
   523→      });
   524→
   525→      if (!error) {
   526→        const success = data as boolean;
   527→        if (success) {
   528→          setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   529→        }
   530→        return success;
   531→      }
   532→
   533→      // RPC doesn't exist or failed — proceed to fallback strategy.
   534→      if (error.message.includes('Could not find the function') || error.code === 'PGRST202') {
   535→        console.warn('[AppContext] activate_subscription RPC not found — falling back to direct UPDATE.');
   536→      } else {
   537→        console.warn('[AppContext] activate_subscription RPC error:', error.message, '— falling back to direct UPDATE.');
   538→      }
   539→    } catch (err) {
   540→      console.warn('[AppContext] activate_subscription RPC error:', err, '— falling back to direct UPDATE.');
   541→    }
   542→
   543→    // ── Strategy 2: direct PostgREST UPDATE ──
   544→    // Company owners have `companies_owner` RLS policy which allows
   545→    // UPDATE on their own row. This persists activation to the DB.
   546→    const companyId = authCompanyId;
   547→    if (companyId) {
   548→      try {
   549→        const { error: updateError } = await supabase
   550→          .from('companies')
   551→          .update({ subscription_active: true })
   552→          .eq('id', companyId);
   553→
   554→        if (!updateError) {
   555→          setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   556→          return true;
   557→        }
   558→
   559→        console.warn('[AppContext] direct UPDATE failed:', updateError.message);
   560→      } catch (err) {
   561→        console.warn('[AppContext] direct UPDATE error:', err);
   562→      }
   563→    }
   564→
   565→    // ── Strategy 3: both strategies failed — activation not persisted ──
   566→    // Return false so the user sees an error and can retry.
   567→    return false;
   568→  };
   569→
   570→  // ── Auth actions ──────────────────────────────────────────────────────────
   571→
   572→  const logout = () => {
   573→    void signOut().then(() => setLocation('/'));
   574→  };
   575→
   576→  const logoutDriver = () => {
   577→    void signOut().then(() => setLocation('/'));
   578→  };
   579→
   580→  // ── Driver profile ────────────────────────────────────────────────────────
   581→
   582→  const updateDriverProfile = (
   583→    data: Partial<Pick<Driver, 'name' | 'email' | 'vehicleNumber' | 'profilePictureUrl'>>
   584→  ) => {
   585→    if (!currentDriverId) return;
   586→    setDrivers((prev) =>
   587→      prev.map((d) => (d.id === currentDriverId ? { ...d, ...data } : d))
   588→    );
   589→    void updateDriver(currentDriverId, data);
   590→  };
   591→
   592→  // ── Load management ───────────────────────────────────────────────────────
   593→
   594→  const upsertLoad = (input: {
   595→    id?: string;
   596→    productName: string;
   597→    quantity: number;
   598→    unitPrice: number;
   599→  }) => {
   600→    if (!currentDriverId) return;
   601→    const productName = input.productName.trim();
   602→
   603→    // Capture pre-mutation cargo so the finalize call below freezes yesterday's
   604→    // EOD state, not the post-edit state. UI updates synchronously via setLoads.
   605→    const preMutationCargo = loads.filter((l) => l.driverId === currentDriverId);
   606→
   607→    setLoads((prev) => {
   608→      const existingIdx = prev.findIndex(
   609→        (l) =>
   610→          l.driverId === currentDriverId &&
   611→          l.productName.trim().toLowerCase() === productName.toLowerCase()
   612→      );
   613→
   614→      if (existingIdx !== -1) {
   615→        const updated = [...prev];
   616→        const existing = { ...updated[existingIdx], quantity: input.quantity, unitPrice: input.unitPrice };
   617→        updated[existingIdx] = existing;
   618→
   619→        void dbUpsertLoad({
   620→          id: existing.id,
   621→          driverId: currentDriverId,
   622→          productName: existing.productName,
   623→          quantity: existing.quantity,
   624→          unitPrice: existing.unitPrice,
   625→          isNew: false,
   626→        });
   627→
   628→        return updated;
   629→      }
   630→
   631→      const newId = input.id ?? generateId();
   632→      const newItem: CargoItem = {
   633→        id: newId,
   634→        driverId: currentDriverId,
   635→        productName,
   636→        quantity: input.quantity,
   637→        unitPrice: input.unitPrice,
   638→      };
   639→
   640→      void dbUpsertLoad({
   641→        id: newId,
   642→        driverId: currentDriverId,
   643→        productName,
   644→        quantity: input.quantity,
   645→        unitPrice: input.unitPrice,
   646→        isNew: true,
   647→      });
   648→
   649→      return [...prev, newItem];
   650→    });
   651→
   652→    // Finalize yesterday's snapshot AFTER local state update, using captured
   653→    // pre-mutation state. Fire-and-forget — idempotent.
   654→    void finalizeYesterdayIfNeeded(currentDriverId, preMutationCargo);
   655→
   656→    // Latch the "cargo edited today" flag so the Day-2 carry-over title
   657→    // immediately flips from "الحمولة المتبقية من اليوم السابق" to
   658→    // "الحمولة الحالية". See the type-doc on `cargoEditedToday`.
   659→    markCargoEditedToday();
   660→  };
   661→
   662→  const removeLoad = (id: string) => {
   663→    if (!currentDriverId) return;
   664→    // Capture pre-mutation cargo for the finalize call below.
   665→    const preMutationCargo = loads.filter((l) => l.driverId === currentDriverId);
   666→    setLoads((prev) => prev.filter((l) => l.id !== id));
   667→    void dbRemoveLoad(id);
   668→    void finalizeYesterdayIfNeeded(currentDriverId, preMutationCargo);
   669→    // Latch the "cargo edited today" flag — removing a product is an edit.
   670→    markCargoEditedToday();
   671→  };
   672→
   673→  // ── Historical snapshot → live cargo promotion ─────────────────────────────
   674→  //
   675→  // Per spec: when the driver edits any product from a past day's "Remaining
   676→  // Cargo From This Day", that snapshot immediately becomes the live Current
   677→  // Cargo. We:
   678→  //   1. Fetch the snapshot for the given date.
   679→  //   2. Filter out qty-0 items (already hidden in the UI, but defensive).
   680→  //   3. Assign fresh UUIDs to each item (new live rows, not snapshot rows).
   681→  //   4. Optimistically replace the driver's loads in local state.
   682→  //   5. Atomically replace the driver's loads in the DB.
   683→  //   6. Return the new live CargoItem[] so the caller can find the matching
   684→  //      item and open the Load tab with it prefilled.
   685→  //
   686→  // Note: this is destructive — the driver's previous live loads are lost.
   687→  // This is the intended behavior per spec ("It immediately becomes Current
   688→  // Cargo because it is now the active working inventory again").
   689→  //
   690→  // SAFETY: if the snapshot for `snapshotDate` does not exist (or is empty),
   691→  // we ABORT the promotion and return the current live cargo unchanged.
   692→  // Without this guard, a missing snapshot would cause `setLoads` to delete
   693→  // ALL of the driver's current live cargo — a destructive no-op. See Bug 4
   694→  // in the worklog.
   695→  const promoteSnapshotToLive = async (
   696→    driverId: string,
   697→    snapshotDate: string
   698→  ): Promise<CargoItem[]> => {
   699→    const snapshotItems = await fetchDailySnapshots([driverId], snapshotDate);
   700→    const itemsToPromote = snapshotItems.filter((item) => item.quantity > 0);
   701→
   702→    // SAFETY: no snapshot (or all-qty-0 snapshot) → abort promotion.
   703→    // Return the driver's CURRENT live cargo unchanged so the caller can
   704→    // still open the editor against the matching product. Without this
   705→    // guard, a missing snapshot would cause `setLoads` to delete ALL of
   706→    // the driver's current live cargo — a destructive no-op. See Bug 4
   707→    // in the worklog.
   708→    if (itemsToPromote.length === 0) {
   709→      return loads.filter((l) => l.driverId === driverId);
   710→    }
   711→
   712→    const newCargo: CargoItem[] = itemsToPromote.map((item) => ({
   713→      id: generateId(),
   714→      driverId,
   715→      productName: item.productName,
   716→      quantity: item.quantity,
   717→      unitPrice: item.unitPrice,
   718→    }));
   719→
   720→    // Optimistic local state update: remove old loads for this driver, add new.
   721→    setLoads((prev) => [
   722→      ...prev.filter((l) => l.driverId !== driverId),
   723→      ...newCargo,
   724→    ]);
   725→
   726→    // DB: atomically replace the driver's loads.
   727→    await replaceDriverLoads(driverId, newCargo);
   728→
   729→    // Promoting a historical snapshot to live IS a cargo edit — latch the
   730→    // "cargo edited today" flag so today's title becomes "الحمولة الحالية".
   731→    // Only latches when the promotion targets the current driver (the
   732→    // company-dashboard case where `driverId === currentDriverId`); the
   733→    // company dashboard is read-only and never calls this with another
   734→    // driver's id, but we guard anyway for safety.
   735→    if (driverId === currentDriverId) {
   736→      markCargoEditedToday();
   737→    }
   738→
   739→    return newCargo;
   740→  };
   741→
   742→  // ── Sales ─────────────────────────────────────────────────────────────────
   743→
   744→  const addSale = (items: SaleLineItem[], receiptImageUrl?: string | null) => {
   745→    if (!currentDriverId) return;
   746→
   747→    // Capture pre-mutation cargo so the finalize call below freezes yesterday's
   748→    // EOD state, not the post-sale (decremented) state. UI updates synchronously.
   749→    const preMutationCargo = loads.filter((l) => l.driverId === currentDriverId);
   750→
   751→    setLoads((prev) =>
   752→      prev.map((load) => {
   753→        if (load.driverId !== currentDriverId) return load;
   754→        const soldItem = items.find(
   755→          (i) => i.productName.trim().toLowerCase() === load.productName.trim().toLowerCase()
   756→        );
   757→        if (!soldItem) return load;
   758→        const newQty = Math.max(0, load.quantity - soldItem.quantity);
   759→        void decrementLoad(currentDriverId, load.productName, soldItem.quantity);
   760→        return { ...load, quantity: newQty };
   761→      })
   762→    );
   763→
   764→    // Finalize yesterday's snapshot AFTER local state update, using captured
   765→    // pre-mutation state. Fire-and-forget — idempotent. Must come before the
   766→    // createSale call so yesterday's EOD is frozen before today's first sale
   767→    // is recorded, but it doesn't need to be awaited (the snapshot insert is
   768→    // independent of the sale insert; both are async DB writes).
   769→    void finalizeYesterdayIfNeeded(currentDriverId, preMutationCargo);
   770→
   771→    const totalPrice = items.reduce((sum, i) => sum + i.quantity * i.unitPrice, 0);
   772→    // Baghdad local date (UTC+3, no DST). en-CA → 'YYYY-MM-DD'.
   773→    // Previously used UTC which mis-stamped sales made between 00:00–03:00 Baghdad.
   774→    const date = new Date().toLocaleDateString('en-CA', { timeZone: 'Asia/Baghdad' });
   775→
   776→    const newSale: SaleRecord = {
   777→      id: generateId(),
   778→      driverId: currentDriverId,
   779→      date,
   780→      items,
   781→      totalPrice,
   782→      receiptImageUrl: receiptImageUrl ?? null,
   783→    };
   784→    setSales((prev) => [newSale, ...prev]);
   785→    void createSale(newSale.id, currentDriverId, date, items, totalPrice, receiptImageUrl ?? null);
   786→
   787→    // ── Notify company owner via FCM Edge Function ────────────────────────
   788→    // The driver's app (which is open) calls the Edge Function, which sends
   789→    // an FCM Web Push to the company owner's device. This works even when
   790→    // the company app/PWA is completely closed.
   791→    //
   792→    // IMPORTANT: For drivers, authCompanyId is NULL (AuthContext only sets it
   793→    // for company-role users). The driver's companyId comes from their profile
   794→    // (authDriverProfile.companyId), which is populated from the drivers table's
   795→    // company_id foreign key during bootstrap.
   796→    const driverCompanyId = role === 'driver'
   797→      ? authDriverProfile?.companyId
   798→      : authCompanyId;
   799→    if (isFirebaseConfigured && role === 'driver' && driverCompanyId) {
   800→      const driverName = driversRef.current.find((d) => d.id === currentDriverId)?.name || '';
   801→      void notifySaleViaEdgeFunction(
   802→        newSale.id,
   803→        currentDriverId,
   804→        driverName,
   805→        totalPrice,
   806→        driverCompanyId,
   807→      );
   808→    }
   809→
   810→    // Per the revised midnight-logic spec, a sale IS a cargo mutation —
   811→    // it decrements live quantities. Latch the "cargo edited today" flag
   812→    // so the Day-2 carry-over title flips from "الحمولة المتبقية من اليوم
   813→    // السابق" to "الحمولة الحالية" on the first sale of the day.
   814→    markCargoEditedToday();
   815→  };
   816→
   817→  // When enabled: request permission + register FCM token + save to Supabase.
   818→  // If token registration FAILS, the toggle flips back OFF — deterministic.
   819→  const enableNotifications = async () => {
   820→    if (typeof window === 'undefined' || !('Notification' in window)) {
   821→      setNotificationPermission('unsupported');
   822→      return;
   823→    }
   824→
   825→    // Request browser notification permission
   826→    const permission = await fcmRequestPermission();
   827→    setNotificationPermission(permission);
   828→
   829→    if (permission === 'granted') {
   830→      // Optimistically enable — will revert if token registration fails
   831→      setNotificationsEnabled(true);
   832→      localStorage.setItem(NOTIFICATIONS_STORAGE_KEY, '1');
   833→
   834→      // Register FCM token — AWAIT the result (not fire-and-forget)
   835→      if (isFirebaseConfigured && role === 'company' && authCompanyId) {
   836→        const token = await registerFcmToken(authCompanyId);
   837→        if (!token) {
   838→          // Token registration failed — revert toggle to OFF
   839→          setNotificationsEnabled(false);
   840→          localStorage.setItem(NOTIFICATIONS_STORAGE_KEY, '0');
   841→        }
   842→      }
   843→    } else {
   844→      // Denied or dismissed — keep the toggle off.
   845→      setNotificationsEnabled(false);
   846→      localStorage.setItem(NOTIFICATIONS_STORAGE_KEY, '0');
   847→    }
   848→  };
   849→
   850→  // When disabled: remove FCM token from Supabase (stop push delivery).
   851→  const disableNotifications = () => {
   852→    setNotificationsEnabled(false);
   853→    localStorage.setItem(NOTIFICATIONS_STORAGE_KEY, '0');
   854→
   855→    // Remove FCM token so the Edge Function stops sending push to this device
   856→    if (isFirebaseConfigured && role === 'company' && authCompanyId) {
   857→      void unregisterFcmToken(authCompanyId);
   858→    }
   859→  };
   860→
   861→  // ── FCM foreground message handler ──────────────────────────────────────
   862→  // When the page IS in the foreground, FCM delivers messages to the
   863→  // onMessage callback instead of the service worker's onBackgroundMessage.
   864→  // We show a notification via SW's showNotification() (not new Notification)
   865→  // to keep it branded as "TrackTracker" and avoid the Chrome/TrackTracker
   866→  // duplicate issue.
   867→  useEffect(() => {
   868→    if (role !== 'company' || !authCompanyId) return;
   869→    if (!isFirebaseConfigured || !notificationsEnabled) return;
   870→
   871→    let unsubscribe: (() => void) | null = null;
   872→
   873→    void (async () => {
   874→      const messaging = await getFirebaseMessaging();
   875→      if (!messaging) return;
   876→
   877→      const { onMessage } = await import('firebase/messaging');
   878→
   879→      unsubscribe = onMessage(messaging, (payload) => {
   880→        const data = payload.data || {};
   881→        const saleId = data.saleId || '';
   882→        const type = data.type || '';
   883→
   884→        // Only handle sale notifications
   885→        if (type !== 'sale') return;
   886→
   887→        // Dedup guard — skip if we already notified for this sale
   888→        const now = Date.now();
   889→        const DEDUP_TTL = 5 * 60 * 1000;
   890→        const dedupMap = notifiedSaleIdsRef.current;
   891→        if (saleId && dedupMap.has(saleId)) {
   892→          return;
   893→        }
   894→        // Expire old entries
   895→        for (const [id, ts] of dedupMap) {
   896→          if (now - ts > DEDUP_TTL) dedupMap.delete(id);
   897→        }
   898→        if (saleId) dedupMap.set(saleId, now);
   899→
   900→        const title = data.title || 'عملية بيع جديدة';
   901→        const body = data.body || 'تم تسجيل عملية بيع جديدة';
   902→        const icon = data.icon || 'https://track-tracker-app.vercel.app/icons/icon-192.png';
   903→        const badge = 'https://track-tracker-app.vercel.app/icons/notification-bell.png';
   904→        // ALWAYS use sale-based tag for dedup — never Date.now()
   905→        const tag = `sale-${saleId}`;
   906→
   907→        // Use SW showNotification() — single "TrackTracker"-branded notification
   908→        if ('serviceWorker' in navigator && navigator.serviceWorker.controller) {
   909→          navigator.serviceWorker.ready.then((reg) => {
   910→            reg.showNotification(title, { body, icon, badge, tag, data: { saleId, type, clickAction: '/' }, dir: 'rtl', lang: 'ar' });
   911→          }).catch(() => {
   912→            try { new Notification(title, { body, icon, tag }); } catch {}
   913→          });
   914→        } else {
   915→          try { new Notification(title, { body, icon, tag }); } catch {}
   916→        }
   917→      });
   918→    })();
   919→
   920→    return () => { unsubscribe?.(); };
   921→  }, [role, authCompanyId, notificationsEnabled]);
   922→
   923→  // ── FCM token refresh on visibility change ────────────────────────────────
   924→  // When the app regains focus (e.g., user returns to the tab), re-register
   925→  // the FCM token to handle token rotation by the browser/Firebase.
   926→  useEffect(() => {
   927→    if (role !== 'company' || !authCompanyId) return;
   928→    if (!isFirebaseConfigured || !notificationsEnabled) return;
   929→
   930→    const onVisibilityChange = () => {
   931→      if (document.visibilityState === 'visible') {
   932→        void refreshFcmTokenIfNeeded(authCompanyId);
   933→      }
   934→    };
   935→
   936→    document.addEventListener('visibilitychange', onVisibilityChange);
   937→    return () => document.removeEventListener('visibilitychange', onVisibilityChange);
   938→  }, [role, authCompanyId, notificationsEnabled]);
   939→
   940→  // ── Re-register FCM token on login/refresh ──────────────────────────────
   941→  // When a company owner logs in or refreshes, ensure their FCM token is
   942→  // still registered (handles token expiry, browser update, etc.).
   943→  // Also fires when notificationsEnabled flips to true (user enables toggle).
   944→  useEffect(() => {
   945→    if (role !== 'company' || !authCompanyId) return;
   946→    if (!isFirebaseConfigured || !notificationsEnabled) return;
   947→    if (fcmGetPermission() !== 'granted') return;
   948→
   949→    // Re-register token on mount (handles refresh/login)
   950→    void registerFcmToken(authCompanyId);
   951→  }, [role, authCompanyId, notificationsEnabled]);
   952→
   953→  // ── Supabase Realtime for UI updates ONLY (no notifications) ─────────────
   954→  // Realtime is still used to update the sales list in real-time, but
   955→  // it does NOT display notifications — FCM handles all notifications.
   956→  useEffect(() => {
   957→    if (role !== 'company' || !authCompanyId) return;
   958→    if (!isSupabaseConfigured) return;
   959→
   960→    const channel = supabase
   961→      .channel(`company-sales-realtime-${authCompanyId}`)
   962→      .on(
   963→        'postgres_changes',
   964→        { event: 'INSERT', schema: 'public', table: 'sales' },
   965→        (payload) => {
   966→          // Intentionally NOT showing a notification here.
   967→          // FCM + Edge Function handles all sale notifications.
   968→          // Realtime is only for live UI updates (the sales list refreshes
   969→          // automatically when a new sale is inserted).
   970→          console.log('[AppContext] Realtime: new sale detected (UI update only, no notification)');
   971→        },
   972→      )
   973→      .subscribe();
   974→
   975→    return () => {
   976→      void supabase.removeChannel(channel);
   977→    };
   978→  }, [role, authCompanyId]);
   979→
   980→  const currentDriver = drivers.find((d) => d.id === currentDriverId) ?? null;
   981→
   982→  return (
   983→    <AppContext.Provider
   984→      value={{
   985→        company,
   986→        darkMode,
   987→        toggleDarkMode,
   988→        updateLogo,
   989→        setCompanyProfile,
   990→        setJoinCode,
   991→        regenerateJoinCode,
   992→        logout,
   993→        activateSubscription,
   994→        companySubscriptionActive: role === 'driver'
   995→          ? driverCompanySubscriptionActive
   996→          : company.subscriptionActive,
   997→        drivers,
   998→        loads,
   999→        sales,
  1000→        currentDriverId,
  1001→        currentDriver,
  1002→        logoutDriver,
  1003→        updateDriverProfile,
  1004→        upsertLoad,
  1005→        removeLoad,
  1006→        promoteSnapshotToLive,
  1007→        addSale,
  1008→        cargoEditedToday,
  1009→        notificationsEnabled,
  1010→        notificationPermission,
  1011→        enableNotifications,
  1012→        disableNotifications,
  1013→        fcmRegStatus,
  1014→        fcmRegError,
  1015→        fcmDiagnostics: fcmDiagnosticsState,
  1016→        sendTestFcmNotification: async () => {
  1017→          if (!authCompanyId) return { success: false, message: 'Not logged in' };
  1018→          return sendTestNotification(authCompanyId);
  1019→        },
  1020→      }}
  1021→    >
  1022→      {children}
  1023→    </AppContext.Provider>
  1024→  );
  1025→}
  1026→
  1027→export function useApp(): AppContextType {
  1028→  const ctx = useContext(AppContext);
  1029→  if (!ctx) throw new Error('useApp must be used inside <AppProvider>');
  1030→  return ctx;
  1031→}
