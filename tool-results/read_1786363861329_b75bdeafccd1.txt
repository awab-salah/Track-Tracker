     1→/**
     2→ * FCM Service — Firebase Cloud Messaging token management for TrackTracker
     3→ *
     4→ * Responsibilities:
     5→ * - Request notification permission
     6→ * - Get/register FCM token bound to the actual PWA Service Worker
     7→ * - Save token to Supabase `fcm_tokens` table (linked to company)
     8→ * - Remove token when notifications are disabled
     9→ * - Handle token refresh
    10→ * - Expose FULL diagnostics state for visible UI panel (no DevTools needed)
    11→ * - Send test notifications for end-to-end verification

    17→// ── Diagnostics State ─────────────────────────────────────────────────────────
    18→// Every step of the FCM registration pipeline is tracked so the UI can show
    19→// exactly what's working and what's broken — no DevTools needed.
    20→
    21→export interface FcmDiagnostics {
    22→  /** Browser Notification.permission */
    23→  browserPermission: NotificationPermission | 'unsupported';
    24→  /** Service Worker: 'registered' | 'activated' | 'missing' | 'unsupported' */
    25→  swStatus: 'registered' | 'activated' | 'missing' | 'unsupported';
    26→  /** PushSubscription: 'created' | 'missing' | 'unsupported' */
    27→  pushSubscription: 'created' | 'missing' | 'unsupported';
    28→  /** FCM token from getToken(): 'registered' | 'missing' */
    29→  fcmToken: 'registered' | 'missing';
    30→  /** Token in fcm_tokens DB: 'saved' | 'missing' | 'error' */
    31→  dbToken: 'saved' | 'missing' | 'error';
    32→  /** Human-readable last error */
    33→  lastError: string;
    34→  /** ISO timestamp of last successful registration */
    35→  lastSuccessTime: string;
    36→  /** High-level registration status */
    37→  regStatus: 'idle' | 'requesting' | 'registered' | 'failed' | 'unregistered';
    38→  /** The actual FCM token string (for diagnostics, truncated) */
    39→  tokenPreview: string;
    40→}

    80→// ── Service Worker Registration ────────────────────────────────────────────────
    81→let swRegistration: ServiceWorkerRegistration | null = null;
    82→
    83→/**
    84→ * Get the active service worker registration with timeout and retry.
    85→ * Updates diagnostics.swStatus at each step.
    86→ */
    87→async function getSwRegistration(): Promise<ServiceWorkerRegistration | null> {
    88→  if (!('serviceWorker' in navigator)) {
    89→    updateDiagnostics({ swStatus: 'unsupported' });
    90→    return null;
    91→  }
    92→
    93→  if (swRegistration) {
    94→    if (swRegistration.active) {
    95→      updateDiagnostics({ swStatus: 'activated' });
    96→      return swRegistration;
    97→    }
    98→    swRegistration = null;
    99→  }
   100→
   101→  const MAX_RETRIES = 3;
   102→  const TIMEOUT_MS = 10_000;
   103→
   104→  for (let attempt = 1; attempt <= MAX_RETRIES; attempt++) {
   105→    try {
   106→      const registration = await Promise.race([
   107→        navigator.serviceWorker.ready,
   108→        new Promise<null>((resolve) =>
   109→          setTimeout(() => resolve(null), TIMEOUT_MS)
   110→        ),
   111→      ]);
   112→
   113→      if (registration && registration.active) {
   114→        swRegistration = registration;
   115→        updateDiagnostics({ swStatus: 'activated' });
   116→        return swRegistration;
   117→      }
   118→
   119→      // SW registered but not yet active
   120→      if (registration) {
   121→        updateDiagnostics({ swStatus: 'registered' });
   122→      }
   123→    } catch (err) {
   124→      updateDiagnostics({
   125→        swStatus: 'missing',
   126→        lastError: `SW registration attempt ${attempt} failed: ${err instanceof Error ? err.message : String(err)}`,
   127→      });
   128→    }
   129→
   130→    if (attempt < MAX_RETRIES) {
   131→      await new Promise((r) => setTimeout(r, attempt * 2000));
   132→    }
   133→  }
   134→
   135→  updateDiagnostics({ swStatus: 'missing' });
   136→  return null;
   137→}

   148→// ── Permission ────────────────────────────────────────────────────────────────

   150→export async function requestNotificationPermission(): Promise<NotificationPermission | 'unsupported'> {
   151→  if (typeof window === 'undefined' || !('Notification' in window)) {
   152→    updateDiagnostics({ browserPermission: 'unsupported' });
   153→    return 'unsupported';
   154→  }
   155→
   156→  let permission = Notification.permission;
   157→  if (permission === 'default') {
   158→    permission = await Notification.requestPermission();
   159→  }
   160→  updateDiagnostics({ browserPermission: permission });
   161→  return permission;
   162→}
   163→
   164→export function getNotificationPermission(): NotificationPermission | 'unsupported' {
   165→  if (typeof window === 'undefined' || !('Notification' in window)) {
   166→    return 'unsupported';
   167→  }
   168→  const p = Notification.permission;
   169→  updateDiagnostics({ browserPermission: p });
   170→  return p;
   171→}

   173→// ── Refresh Diagnostics ──────────────────────────────────────────────────────
   174→/**
   175→ * Probe the browser's current state and update diagnostics.
   176→ * Call this on mount and periodically.
   177→ */
   178→export async function refreshFcmDiagnostics(): Promise<FcmDiagnostics> {
   179→  // Browser permission
   180→  const perm = typeof window !== 'undefined' && 'Notification' in window
   181→    ? Notification.permission
   182→    : 'unsupported' as const;
   183→  const patch: Partial<FcmDiagnostics> = { browserPermission: perm };
   184→
   185→  // Service Worker status
   186→  if ('serviceWorker' in navigator) {
   187→    try {
   188→      const reg = await navigator.serviceWorker.getRegistration();
   189→      if (reg?.active) {
   190→        patch.swStatus = 'activated';
   191→      } else if (reg) {
   192→        patch.swStatus = 'registered';
   193→      } else {
   194→        patch.swStatus = 'missing';
   195→      }
   196→
   197→      // Push subscription
   198→      if (reg) {
   199→        const sub = await reg.pushManager.getSubscription();
   200→        patch.pushSubscription = sub ? 'created' : 'missing';
   201→      } else {
   202→        patch.pushSubscription = 'missing';
   203→      }
   204→    } catch {
   205→      patch.swStatus = 'missing';
   206→      patch.pushSubscription = 'unsupported';
   207→    }
   208→  } else {
   209→    patch.swStatus = 'unsupported';
   210→    patch.pushSubscription = 'unsupported';
   211→  }
   212→
   213→  // FCM token status — check if getToken returns a token
   214→  if (isFirebaseConfigured && perm === 'granted') {
   215→    try {
   216→      const messaging = await getFirebaseMessaging();
   217→      if (messaging && swRegistration) {
   218→        const token = await getToken(messaging, {
   219→          vapidKey: VAPID_KEY,
   220→          serviceWorkerRegistration: swRegistration,
   221→        });
   222→        if (token) {
   223→          patch.fcmToken = 'registered';
   224→          patch.tokenPreview = token.substring(0, 20) + '...';
   225→        } else {
   226→          patch.fcmToken = 'missing';
   227→          patch.tokenPreview = '';
   228→        }
   229→      }
   230→    } catch {
   231→      patch.fcmToken = 'missing';
   232→    }
   233→  }
   234→
   235→  updateDiagnostics(patch);
   236→  return diagnostics;
   237→}

   239→// ── Token Management ──────────────────────────────────────────────────────────
   240→
   241→/**
   242→ * Get an FCM registration token and save it to Supabase for the given company.
   243→ *
   244→ * Each step updates diagnostics so the UI shows exactly what's happening.
   245→ * If ANY step fails, the function returns null and diagnostics.lastError
   246→ * contains the exact failure reason.
   247→ *
   248→ * @param companyId - The UUID of the company from the companies table
   249→ * @returns The FCM token string, or null if unavailable
   250→ */
   251→export async function registerFcmToken(companyId: string): Promise<string | null> {
   252→  updateDiagnostics({ regStatus: 'requesting', lastError: '' });
   253→
   254→  // STEP 0: Check Firebase config
   255→  if (!isFirebaseConfigured) {
   256→    updateDiagnostics({ regStatus: 'failed', lastError: 'Firebase not configured — check env vars' });
   257→    return null;
   258→  }
   259→
   260→  // STEP 1: Check Supabase config
   261→  if (!isSupabaseConfigured) {
   262→    updateDiagnostics({ regStatus: 'failed', lastError: 'Supabase not configured — check env vars' });
   263→    return null;
   264→  }
   265→
   266→  // STEP 2: Check notification permission
   267→  const permission = getNotificationPermission();
   268→  if (permission !== 'granted') {
   269→    updateDiagnostics({
   270→      regStatus: 'failed',
   271→      lastError: `Browser notification permission: ${permission}. Grant permission and try again.`,
   272→      browserPermission: permission,
   273→    });
   274→    return null;
   275→  }
   276→
   277→  // STEP 3: Get Firebase Messaging instance
   278→  const messaging = await getFirebaseMessaging();
   279→  if (!messaging) {
   280→    updateDiagnostics({ regStatus: 'failed', lastError: 'Firebase Messaging not supported in this browser' });
   281→    return null;
   282→  }
   283→
   284→  // Retry getToken() up to 3 times
   285→  const MAX_ATTEMPTS = 3;
   286→  for (let attempt = 1; attempt <= MAX_ATTEMPTS; attempt++) {
   287→    try {
   288→      // STEP 4: Get the active PWA Service Worker registration
   289→      const registration = await getSwRegistration();
   290→      if (!registration) {
   291→        if (attempt < MAX_ATTEMPTS) {
   292→          await new Promise((r) => setTimeout(r, attempt * 2000));
   293→          continue;
   294→        }
   295→        updateDiagnostics({
   296→          regStatus: 'failed',
   297→          lastError: 'Service Worker not ready after retries. Try refreshing the page.',
   298→        });
   299→        return null;
   300→      }
   301→
   302→      // STEP 4b: Verify pushManager exists
   303→      if (!registration.pushManager) {
   304→        updateDiagnostics({
   305→          regStatus: 'failed',
   306→          lastError: 'PushManager not available on this Service Worker registration.',
   307→          pushSubscription: 'unsupported',
   308→        });
   309→        return null;
   310→      }
   311→
   312→      // STEP 5: Clean any stale push subscription
   313→      await cleanStalePushSubscription(registration);
   314→
   315→      // STEP 6: Get FCM token bound to our actual SW
   316→      const token = await getToken(messaging, {
   317→        vapidKey: VAPID_KEY,
   318→        serviceWorkerRegistration: registration,
   319→      });
   320→
   321→      if (!token) {
   322→        if (attempt < MAX_ATTEMPTS) {
   323→          await new Promise((r) => setTimeout(r, attempt * 2000));
   324→          continue;
   325→        }
   326→        updateDiagnostics({
   327→          regStatus: 'failed',
   328→          lastError: 'getToken() returned empty. Push subscription may be invalid.',
   329→          fcmToken: 'missing',
   330→        });
   331→        return null;
   332→      }
   333→
   334→      // Token obtained!
   335→      updateDiagnostics({
   336→        fcmToken: 'registered',
   337→        tokenPreview: token.substring(0, 20) + '...',
   338→      });
   339→
   340→      // STEP 7: Check push subscription exists
   341→      const sub = await registration.pushManager.getSubscription();
   342→      updateDiagnostics({ pushSubscription: sub ? 'created' : 'missing' });
   343→
   344→      // STEP 8: Save token to Supabase
   345→      const saveResult = await saveTokenToSupabase(companyId, token);
   346→      if (!saveResult) {
   347→        updateDiagnostics({
   348→          regStatus: 'failed',
   349→          lastError: `Failed to save FCM token to database. This is likely an RLS (Row Level Security) issue — the fcm_tokens table may not allow INSERT for the anon key. Error: ${diagnostics.lastError}`,
   350→          dbToken: 'error',
   351→        });
   352→        return null;
   353→      }
   354→
   355→      // STEP 9: Verify the token is actually in the database
   356→      const verified = await verifyTokenInDb(companyId, token);
   357→      if (!verified) {
   358→        updateDiagnostics({
   359→          regStatus: 'failed',
   360→          lastError: 'Token was saved but could not be verified in database. RLS may be blocking SELECT.',
   361→          dbToken: 'error',
   362→        });
   363→        return null;
   364→      }
   365→
   366→      // SUCCESS!
   367→      updateDiagnostics({
   368→        regStatus: 'registered',
   369→        dbToken: 'saved',
   370→        lastError: '',
   371→        lastSuccessTime: new Date().toISOString(),
   372→      });
   373→
   374→      return token;
   375→    } catch (err: unknown) {
   376→      const errMsg = err instanceof Error ? err.message : String(err);
   377→      const errCode = (err as { code?: string })?.code ?? '';
   378→
   379→      // Permission errors are not retryable
   380→      if (errCode === 'messaging/permission-blocked' || errMsg.includes('permission')) {
   381→        updateDiagnostics({
   382→          regStatus: 'failed',
   383→          lastError: `Permission error: ${errMsg}`,
   384→        });
   385→        return null;
   386→      }
   387→
   388→      if (attempt < MAX_ATTEMPTS) {
   389→        updateDiagnostics({ lastError: `Attempt ${attempt} failed (${errCode}): ${errMsg}. Retrying...` });
   390→        await new Promise((r) => setTimeout(r, attempt * 3000));
   391→      } else {
   392→        updateDiagnostics({
   393→          regStatus: 'failed',
   394→          lastError: `${errCode || 'Error'}: ${errMsg}`,
   395→        });
   396→      }
   397→    }
   398→  }
   399→
   400→  return null;
   401→}

   403→/**
   404→ * Remove the FCM token from both Firebase and Supabase.
   405→ * Called when the company owner disables notifications.
   406→ */
   407→export async function unregisterFcmToken(companyId: string): Promise<void> {
   408→  updateDiagnostics({ regStatus: 'unregistered' });
   409→
   410→  if (!isFirebaseConfigured) return;
   411→
   412→  const messaging = await getFirebaseMessaging();
   413→  if (!messaging) return;
   414→
   415→  try {
   416→    await deleteToken(messaging);
   417→  } catch {
   418→    // Token may already be deleted
   419→  }
   420→
   421→  await removeTokenFromSupabase(companyId);
   422→  swRegistration = null;
   423→
   424→  updateDiagnostics({
   425→    fcmToken: 'missing',
   426→    dbToken: 'missing',
   427→    tokenPreview: '',
   428→    pushSubscription: 'missing',
   429→  });
   430→}

   432→/**
   433→ * Re-register FCM token if needed (called on visibility change).
   434→ */
   435→export async function refreshFcmTokenIfNeeded(companyId: string): Promise<void> {
   436→  if (!isFirebaseConfigured || !isSupabaseConfigured) return;
   437→  if (getNotificationPermission() !== 'granted') return;
   438→
   439→  const messaging = await getFirebaseMessaging();
   440→  if (!messaging) return;
   441→
   442→  try {
   443→    const registration = await getSwRegistration();
   444→    if (!registration) return;
   445→
   446→    const token = await getToken(messaging, {
   447→      vapidKey: VAPID_KEY,
   448→      serviceWorkerRegistration: registration,
   449→    });
   450→    if (token) {
   451→      const saved = await saveTokenToSupabase(companyId, token);
   452→      if (saved) {
   453→        updateDiagnostics({
   454→          regStatus: 'registered',
   455→          fcmToken: 'registered',
   456→          dbToken: 'saved',
   457→          tokenPreview: token.substring(0, 20) + '...',
   458→          lastSuccessTime: new Date().toISOString(),
   459→        });
   460→      }
   461→    }
   462→  } catch {
   463→    // Non-critical — will retry on next visibility change
   464→  }
   465→}

   467→// ── Push Subscription Cleanup ─────────────────────────────────────────────────
   468→
   469→async function cleanStalePushSubscription(
   470→  registration: ServiceWorkerRegistration,
   471→): Promise<void> {
   472→  try {
   473→    const subscription = await registration.pushManager.getSubscription();
   474→    if (!subscription) return;
   475→
   476→    const subKey = subscription.options?.applicationServerKey;
   477→    if (subKey) {
   478→      const subKeyB64 = arrayBufferToBase64(
   479→        subKey instanceof ArrayBuffer
   480→          ? subKey
   481→          : new TextEncoder().encode(subKey as string).buffer as ArrayBuffer,
   482→      );
   483→      const vapidB64 = arrayBufferToBase64(base64UrlToArrayBuffer(VAPID_KEY));
   484→
   485→      if (subKeyB64 !== vapidB64) {
   486→        await subscription.unsubscribe();
   487→        updateDiagnostics({ lastError: 'Cleaned stale push subscription with mismatched VAPID key' });
   488→      }
   489→    }
   490→  } catch (err) {
   491→    updateDiagnostics({ lastError: `Stale push cleanup error: ${err instanceof Error ? err.message : String(err)}` });
   492→  }
   493→}
   494→
   495→function arrayBufferToBase64(buffer: ArrayBuffer): string {
   496→  const bytes = new Uint8Array(buffer);
   497→  let binary = '';
   498→  for (let i = 0; i < bytes.byteLength; i++) {
   499→    binary += String.fromCharCode(bytes[i]);
   500→  }
   501→  return btoa(binary);
   502→}
   503→
   504→function base64UrlToArrayBuffer(base64url: string): ArrayBuffer {
   505→  let base64 = base64url.replace(/-/g, '+').replace(/_/g, '/');
   506→  while (base64.length % 4) base64 += '=';
   507→  const binary = atob(base64);
   508→  const bytes = new Uint8Array(binary.length);
   509→  for (let i = 0; i < binary.length; i++) {
   510→    bytes[i] = binary.charCodeAt(i);
   511→  }
   512→  return bytes.buffer as ArrayBuffer;
   513→}

   515→// ── Supabase Token Persistence ────────────────────────────────────────────────
   516→
   517→/**
   518→ * Save or update the FCM token in the `fcm_tokens` table.
   519→ * Returns true on success, false on failure.
   520→ * CRITICAL: Errors are NOT swallowed — they're surfaced in diagnostics.
   521→ */
   522→async function saveTokenToSupabase(companyId: string, token: string): Promise<boolean> {
   523→  // Delete any existing tokens for this company that differ
   524→  const { error: deleteError } = await supabase
   525→    .from('fcm_tokens')
   526→    .delete()
   527→    .eq('company_id', companyId)
   528→    .neq('token', token);
   529→
   530→  if (deleteError) {
   531→    updateDiagnostics({
   532→      dbToken: 'error',
   533→      lastError: `DB delete error (old tokens): ${deleteError.message} (code: ${deleteError.code}, hint: ${deleteError.hint || 'none'})`,
   534→    });
   535→    // Don't return false here — the delete of old tokens is best-effort
   536→  }
   537→
   538→  // Insert the new token
   539→  const { error } = await supabase
   540→    .from('fcm_tokens')
   541→    .upsert(
   542→      { company_id: companyId, token },
   543→      { onConflict: 'company_id,token' }
   544→    );
   545→
   546→  if (error) {
   547→    updateDiagnostics({
   548→      dbToken: 'error',
   549→      lastError: `DB upsert error: ${error.message} (code: ${error.code}, details: ${error.details || 'none'}, hint: ${error.hint || 'none'})`,
   550→    });
   551→    return false;
   552→  }
   553→
   554→  return true;
   555→}

   557→/**
   558→ * Verify that the token actually exists in the database.
   559→ * This catches RLS/permission issues that would silently fail.
   560→ */
   561→async function verifyTokenInDb(companyId: string, token: string): Promise<boolean> {
   562→  const { data, error } = await supabase
   563→    .from('fcm_tokens')
   564→    .select('token')
   565→    .eq('company_id', companyId)
   566→    .eq('token', token)
   567→    .limit(1);
   568→
   569→  if (error) {
   570→    updateDiagnostics({
   571→      dbToken: 'error',
   572→      lastError: `DB verify SELECT error: ${error.message} (code: ${error.code}, hint: ${error.hint || 'none'})`,
   573→    });
   574→    return false;
   575→  }
   576→
   577→  if (!data || data.length === 0) {
   578→    updateDiagnostics({
   579→      dbToken: 'missing',
   580→      lastError: 'Token upsert succeeded but SELECT found no rows — RLS may block SELECT or INSERT silently failed',
   581→    });
   582→    return false;
   583→  }
   584→
   585→  return true;
   586→}

   607→/**
   608→ * Check if a token exists in the database for a given company.
   609→ * Updates diagnostics.dbToken accordingly.
   610→ */
   611→export async function checkDbTokenStatus(companyId: string): Promise<'saved' | 'missing' | 'error'> {
   612→  if (!isSupabaseConfigured) {
   613→    updateDiagnostics({ dbToken: 'error' });
   614→    return 'error';
   615→  }
   616→
   617→  const { data, error } = await supabase
   618→    .from('fcm_tokens')
   619→    .select('token')
   620→    .eq('company_id', companyId)
   621→    .limit(1);
   622→
   623→  if (error) {
   624→    updateDiagnostics({ dbToken: 'error', lastError: `DB check error: ${error.message} (code: ${error.code})` });
   625→    return 'error';
   626→  }
   627→
   628→  const status = data && data.length > 0 ? 'saved' : 'missing';
   629→  updateDiagnostics({ dbToken: status });
   630→  return status;
   631→}

   633→// ── Edge Function Call ────────────────────────────────────────────────────────

   643→/**
   644→ * Call the Supabase Edge Function `notify-sale` using an explicit fetch().
   645→ *
   646→ * We use raw `fetch()` (not `supabase.functions.invoke()`) because:
   647→ * - `supabase.functions.invoke()` wraps the response in a way that can
   648→ *   throw `FunctionsFetchError` / "Failed to send a request to the Edge Function"
   649→ *   even when the Edge Function is reachable and working.
   650→ * - Raw `fetch()` gives us direct control over the exact HTTP status code,
   651→ *   response body, and error semantics — no hidden wrapping.
   652→ *
   653→ * Three headers are required:
   654→ *   1. `apikey`          — Supabase anon key (publishable)
   655→ *   2. `Authorization`   — Bearer <user JWT> from the current session
   656→ *   3. `Content-Type`    — application/json
   657→ */
   658→async function callEdgeFunction(payload: Record<string, unknown>): Promise<{
   659→  ok: boolean;
   660→  status: number;
   661→  body: string;
   662→  data: Record<string, unknown> | null;
   663→}> {
   664→  const { data: { session } } = await supabase.auth.getSession();
   665→  if (!session?.access_token) {
   666→    throw new Error('No authenticated session — please log in again');
   667→  }
   668→
   669→  const url = `${SUPABASE_URL}/functions/v1/notify-sale`;
   670→
   671→  console.log('[fcmService] Edge Function request:', url, payload);
   672→
   673→  const response = await fetch(url, {
   674→    method: 'POST',
   675→    headers: {
   676→      'Content-Type': 'application/json',
   677→      'apikey': SUPABASE_ANON_KEY,
   678→      'Authorization': `Bearer ${session.access_token}`,
   679→    },
   680→    body: JSON.stringify(payload),
   681→  });
   682→
   683→  const bodyText = await response.text();
   684→  console.log('[fcmService] Edge Function response:', response.status, bodyText);
   685→
   686→  let data: Record<string, unknown> | null = null;
   687→  try {
   688→    data = JSON.parse(bodyText);
   689→  } catch {
   690→    // Non-JSON response — that's OK, bodyText still has it
   691→  }
   692→
   693→  return { ok: response.ok, status: response.status, body: bodyText, data };
   694→}

   696→/**
   697→ * Notify the company owner about a new sale via the Supabase Edge Function.
   698→ * Fire-and-forget: logs errors but does not throw.
   699→ */
   700→export async function notifySaleViaEdgeFunction(
   701→  saleId: string,
   702→  driverId: string,
   703→  driverName: string,
   704→  totalPrice: number,
   705→  companyId: string
   706→): Promise<void> {
   707→  if (!isSupabaseConfigured) return;
   708→
   709→  try {
   710→    const result = await callEdgeFunction({
   711→      saleId,
   712→      driverId,
   713→      driverName,
   714→      totalPrice,
   715→      companyId,
   716→      type: 'sale',
   717→    });
   718→
   719→    if (!result.ok) {
   720→      console.error(
   721→        `[fcmService] notify-sale failed: HTTP ${result.status} — ${result.body}`
   722→      );
   723→    }
   724→  } catch (err) {
   725→    console.error('[fcmService] notify-sale exception:', err);
   726→  }
   727→}

   731→/**
   732→ * Send a test FCM notification to THIS device through the production backend.
   733→ * This calls the same Edge Function used for real sale notifications,
   734→ * but with a synthetic "test" sale ID.
   735→ *
   736→ * Returns { success, message } for UI display.
   737→ */
   738→export async function sendTestNotification(companyId: string): Promise<{ success: boolean; message: string }> {
   739→  if (!isSupabaseConfigured) {
   740→    return { success: false, message: 'Supabase not configured' };
   741→  }
   742→
   743→  try {
   744→    // Check if there's a token in the DB first
   745→    const dbStatus = await checkDbTokenStatus(companyId);
   746→    if (dbStatus === 'missing') {
   747→      return { success: false, message: 'No FCM token in database. Enable notifications first, then try again.' };
   748→    }
   749→    if (dbStatus === 'error') {
   750→      return { success: false, message: `Database error: ${diagnostics.lastError}` };
   751→    }
   752→
   753→    const testSaleId = `test-${Date.now()}`;
   754→
   755→    const result = await callEdgeFunction({
   756→      saleId: testSaleId,
   757→      driverId: 'test-driver',
   758→      driverName: 'اختبار',
   759→      totalPrice: 1000,
   760→      companyId,
   761→      type: 'sale',
   762→    });
   763→
   764→    // HTTP-level failure
   765→    if (!result.ok) {
   766→      return {
   767→        success: false,
   768→        message: `Edge Function HTTP ${result.status}: ${result.body.substring(0, 200)}`,
   769→      };
   770→    }
   771→
   772→    // Parse the response
   773→    const r = result.data;
   774→
   775→    if (r?.message && String(r.message).includes('No FCM tokens')) {
   776→      return { success: false, message: 'No FCM tokens registered for this company. Enable notifications first.' };
   777→    }
   778→
   779→    if (r?.error) {
   780→      return { success: false, message: `Server error: ${r.error}` };
   781→    }
   782→
   783→    if (r?.sent && (r.sent as number) > 0) {
   784→      return { success: true, message: `Test notification sent! (${r.sent}/${r.total} devices). Check your notification tray.` };
   785→    }
   786→
   787→    if (r?.sent === 0) {
   788→      return { success: false, message: 'Edge Function returned sent=0. No FCM tokens found for this company.' };
   789→    }
   790→
   791→    return { success: true, message: `Edge Function responded (HTTP ${result.status}): ${result.body.substring(0, 200)}` };
   792→  } catch (err) {
   793→    const msg = err instanceof Error ? err.message : String(err);
   794→    return { success: false, message: `Request failed: ${msg}` };
   795→  }
   796→}
