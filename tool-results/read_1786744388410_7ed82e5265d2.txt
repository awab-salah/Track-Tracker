   126→   126→function generateJoinCode(): string {
   127→   127→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
   128→   128→  return Array.from({ length: 6 }, () => chars[Math.floor(Math.random() * chars.length)]).join('');
   129→   129→}

   131→   131→function generateId(): string {
   132→   132→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
   133→   133→    return crypto.randomUUID();
   134→   134→  }
   135→   135→  return `local-${Date.now().toString(36)}-${Math.floor(Math.random() * 1e4)}`;
   136→   136→}

   146→   146→// Local-only preference — Notification permission itself is a per-browser/device
   147→   147→// concept (not a company-level setting), so it is intentionally not persisted
   148→   148→// to Supabase. Kept in localStorage like the dark-mode toggle.
   149→   149→const NOTIFICATIONS_STORAGE_KEY = 'tt_notifications_enabled';
   150→   150→
   151→   151→/**
   152→   152→ * Build the localStorage key for the "cargo edited today" latch.
   153→   153→ *
   154→   154→ * Keyed by `(driverId, today)` so the flag:
   155→   155→ *   - auto-resets at midnight (today changes → new key → flag starts false),
   156→   156→ *   - is per-driver (multiple drivers on the same browser don't collide),
   157→   157→ *   - survives page refresh (same key → same value).
   158→   158→ *
   159→   159→ * This latch is the mechanism that flips Day-2's cargo title from
   160→   160→ * "الحمولة المتبقية من اليوم السابق" to "الحمولة الحالية" after the driver's
   161→   161→ * FIRST cargo mutation of the day. The carry-over algorithm itself is NOT
   162→   162→ * modified — only the title-resolution layer consumes this flag.
   163→   163→ */
   164→   164→function cargoEditedTodayStorageKey(driverId: string, today: string): string {
   165→   165→  return `tt_cargo_edited_${driverId}_${today}`;
   166→   166→}
   167→   167→
   168→   168→/** Read the latch from localStorage. Returns false if not set or unavailable. */
   169→   169→function readCargoEditedToday(driverId: string, today: string): boolean {
   170→   170→  if (typeof window === 'undefined') return false;
   171→   171→  try {
   172→   172→    return localStorage.getItem(cargoEditedTodayStorageKey(driverId, today)) === '1';
   173→   173→  } catch {
   174→   174→    return false;
   175→   175→  }
   176→   176→}
   177→   177→
   178→   178→/** Write the latch to localStorage. Silently no-op if storage is unavailable. */
   179→   179→function writeCargoEditedToday(driverId: string, today: string, value: boolean): void {
   180→   180→  if (typeof window === 'undefined') return;
   181→   181→  try {
   182→   182→    if (value) {
   183→   183→      localStorage.setItem(cargoEditedTodayStorageKey(driverId, today), '1');
   184→   184→    } else {
   185→   185→      localStorage.removeItem(cargoEditedTodayStorageKey(driverId, today));
   186→   186→    }
   187→   187→  } catch {
   188→   188→    /* ignore storage failures — flag is also kept in React state */
   189→   189→  }
   190→   190→}

   195→   195→}

   197→   197→function dedupeLoads(loads: CargoItem[]): CargoItem[] {
   198→   198→  const byKey = new Map<string, CargoItem>();
   199→   199→  for (const load of loads) {
   200→   200→    const key = `${load.driverId}::${load.productName.trim().toLowerCase()}`;
   201→   201→    const existing = byKey.get(key);
   202→   202→    if (existing) {
   203→   203→      existing.quantity += load.quantity;
   204→   204→      existing.unitPrice = load.unitPrice;
   205→   205→    } else {
   206→   206→      byKey.set(key, { ...load });
   207→   207→    }
   208→   208→  }
   209→   209→  return Array.from(byKey.values());
   210→   210→}

   411→   411→        }
   412→   412→      }

   414→   414→      const [remoteLoads, remoteSales] = await Promise.all([
   415→   415→        fetchLoads([authDriverId]),
   416→   416→        fetchSales([authDriverId]),
   417→   417→      ]);

   419→   419→      setLoads(dedupeLoads(remoteLoads));
   420→   420→      setSales(remoteSales);
   421→   421→
   422→   422→      // Finalize yesterday's snapshot for this driver. Fire-and-forget —
   423→   423→      // idempotent. Reads current DB loads (no mutation today yet).
   424→   424→      void finalizeYesterdayIfNeeded(authDriverId);
   425→   425→    })();
   426→   426→
   427→   427→    return () => { cancelled = true; };
   428→   428→  // eslint-disable-next-line react-hooks/exhaustive-deps
   429→   429→  }, [role, authDriverId, authLoading]);
   430→   430→

   432→   432→  useEffect(() => {
   433→   433→    if (!authLoading && !role) {
   434→   434→      setCompany(DEFAULT_COMPANY);
   435→   435→      setDrivers([]);
   436→   436→      setLoads([]);
   437→   437→      setSales([]);
   438→   438→      setCurrentDriverId(null);
   439→   439→      setDriverCompanySubscriptionActive(false);
   440→   440→    }
   441→   441→  }, [role, authLoading]);

   480→   480→  // ── Subscription activation ──────────────────────────────────────────────
   481→   481→  //
   482→   482→  // Uses the Supabase `activate_subscription` security-definer RPC to validate
   483→   483→  // the activation code and set `subscription_active = true` on the company row.
   484→   484→  // Activation persists ONLY in Supabase — no localStorage, no fallback.
   485→   485→  //
   486→   486→  // The only valid activation code for now is "track1".
   487→   487→  // Later Stripe will replace this — the architecture is ready.
   488→   488→  //
   489→   489→  // Activation strategy (ordered by preference):
   490→   490→  //   1. Call the `activate_subscription` security-definer RPC. This validates
   491→   491→  //      the code server-side and persists to the DB in one atomic call.
   492→   492→  //   2. If the RPC doesn't exist, validate the code locally and do a direct
   493→   493→  //      PostgREST UPDATE on the company row.
   494→   494→  //      This works because company owners have UPDATE RLS on their own row.
   495→   495→  //   3. If both strategies fail, return false (activation not persisted).
   496→   496→  //      The user sees an error and can retry later.
   497→   497→
   498→   498→  const activateSubscription = async (code: string): Promise<boolean> => {
   499→   499→    // If the company is already active, return true immediately.
   500→   500→    // The UI layer checks companySubscriptionActive before calling this
   501→   501→    // function and shows an "already activated" message, but this guard
   502→   502→    // ensures that even if the UI bypass is missed, no duplicate activation
   503→   503→    // attempt reaches the database.
   504→   504→    if (company.subscriptionActive) return true;
   505→   505→
   506→   506→    const normalizedCode = code.trim().toLowerCase();
   507→   507→    const isValid = normalizedCode === 'track1';
   508→   508→
   509→   509→    if (!isValid) return false;
   510→   510→
   511→   511→    if (!isSupabaseConfigured) {
   512→   512→      // Offline mode: accept "track1" and update local state only.
   513→   513→      // No localStorage — state lives only in React (lost on refresh,
   514→   514→      // which is correct since the DB is unreachable in offline mode).
   515→   515→      setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   516→   516→      return true;
   517→   517→    }
   518→   518→
   519→   519→    // ── Strategy 1: try the RPC ──
   520→   520→    try {
   521→   521→      const { data, error } = await supabase.rpc('activate_subscription', {
   522→   522→        p_activation_code: code.trim(),
   523→   523→      });
   524→   524→
   525→   525→      if (!error) {
   526→   526→        const success = data as boolean;
   527→   527→        if (success) {
   528→   528→          setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   529→   529→        }
   530→   530→        return success;
   531→   531→      }
   532→   532→
   533→   533→      // RPC doesn't exist or failed — proceed to fallback strategy.
   534→   534→      if (error.message.includes('Could not find the function') || error.code === 'PGRST202') {
   535→   535→        console.warn('[AppContext] activate_subscription RPC not found — falling back to direct UPDATE.');
   536→   536→      } else {
   537→   537→        console.warn('[AppContext] activate_subscription RPC error:', error.message, '— falling back to direct UPDATE.');
   538→   538→      }
   539→   539→    } catch (err) {
   540→   540→      console.warn('[AppContext] activate_subscription RPC error:', err, '— falling back to direct UPDATE.');
   541→   541→    }
   542→   542→
   543→   543→    // ── Strategy 2: direct PostgREST UPDATE ──
   544→   544→    // Company owners have `companies_owner` RLS policy which allows
   545→   545→    // UPDATE on their own row. This persists activation to the DB.
   546→   546→    const companyId = authCompanyId;
   547→   547→    if (companyId) {
   548→   548→      try {
   549→   549→        const { error: updateError } = await supabase
   550→   550→          .from('companies')
   551→   551→          .update({ subscription_active: true })
   552→   552→          .eq('id', companyId);
   553→   553→
   554→   554→        if (!updateError) {
   555→   555→          setCompany((prev) => ({ ...prev, subscriptionActive: true }));
   556→   556→          return true;
   557→   557→        }
   558→   558→
   559→   559→        console.warn('[AppContext] direct UPDATE failed:', updateError.message);
   560→   560→      } catch (err) {
   561→   561→        console.warn('[AppContext] direct UPDATE error:', err);
   562→   562→      }
   563→   563→    }
   564→   564→
   565→   565→    // ── Strategy 3: both strategies failed — activation not persisted ──
   566→   566→    // Return false so the user sees an error and can retry.
   567→   567→    return false;
   568→   568→  };

   673→   673→  // ── Historical snapshot → live cargo promotion ─────────────────────────────
   674→   674→  //
   675→   675→  // Per spec: when the driver edits any product from a past day's "Remaining
   676→   676→  // Cargo From This Day", that snapshot immediately becomes the live Current
   677→   677→  // Cargo. We:
   678→   678→  //   1. Fetch the snapshot for the given date.
   679→   679→  //   2. Filter out qty-0 items (already hidden in the UI, but defensive).
   680→   680→  //   3. Assign fresh UUIDs to each item (new live rows, not snapshot rows).
   681→   681→  //   4. Optimistically replace the driver's loads in local state.
   682→   682→  //   5. Atomically replace the driver's loads in the DB.
   683→   683→  //   6. Return the new live CargoItem[] so the caller can find the matching
   684→   684→  //      item and open the Load tab with it prefilled.
   685→   685→  //
   686→   686→  // Note: this is destructive — the driver's previous live loads are lost.
   687→   687→  // This is the intended behavior per spec ("It immediately becomes Current
   688→   688→  // Cargo because it is now the active working inventory again").
   689→   689→  //
   690→   690→  // SAFETY: if the snapshot for `snapshotDate` does not exist (or is empty),
   691→   691→  // we ABORT the promotion and return the current live cargo unchanged.
   692→   692→  // Without this guard, a missing snapshot would cause `setLoads` to delete
   693→   693→  // ALL of the driver's current live cargo — a destructive no-op. See Bug 4
   694→   694→  // in the worklog.
   695→   695→  const promoteSnapshotToLive = async (
   696→   696→    driverId: string,
   697→   697→    snapshotDate: string
   698→   698→  ): Promise<CargoItem[]> => {
   699→   699→    const snapshotItems = await fetchDailySnapshots([driverId], snapshotDate);
   700→   700→    const itemsToPromote = snapshotItems.filter((item) => item.quantity > 0);
   701→   701→
   702→   702→    // SAFETY: no snapshot (or all-qty-0 snapshot) → abort promotion.
   703→   703→    // Return the driver's CURRENT live cargo unchanged so the caller can
   704→   704→    // still open the editor against the matching product. Without this
   705→   705→    // guard, a missing snapshot would cause `setLoads` to delete ALL of
   706→   706→    // the driver's current live cargo — a destructive no-op. See Bug 4
   707→   707→    // in the worklog.
   708→   708→    if (itemsToPromote.length === 0) {
   709→   709→      return loads.filter((l) => l.driverId === driverId);
   710→   710→    }
   711→   711→
   712→   712→    const newCargo: CargoItem[] = itemsToPromote.map((item) => ({
   713→   713→      id: generateId(),
   714→   714→      driverId,
   715→   715→      productName: item.productName,
   716→   716→      quantity: item.quantity,
   717→   717→      unitPrice: item.unitPrice,
   718→   718→    }));
   719→   719→
   720→   720→    // Optimistic local state update: remove old loads for this driver, add new.
   721→   721→    setLoads((prev) => [
   722→   722→      ...prev.filter((l) => l.driverId !== driverId),
   723→   723→      ...newCargo,
   724→   724→    ]);
   725→   725→
   726→   726→    // DB: atomically replace the driver's loads.
   727→   727→    await replaceDriverLoads(driverId, newCargo);
   728→   728→
   729→   729→    // Promoting a historical snapshot to live IS a cargo edit — latch the
   730→   730→    // "cargo edited today" flag so today's title becomes "الحمولة الحالية".
   731→   731→    // Only latches when the promotion targets the current driver (the
   732→   732→    // company-dashboard case where `driverId === currentDriverId`); the
   733→   733→    // company dashboard is read-only and never calls this with another
   734→   734→    // driver's id, but we guard anyway for safety.
   735→   735→    if (driverId === currentDriverId) {
   736→   736→      markCargoEditedToday();
   737→   737→    }
   738→   738→
   739→   739→    return newCargo;
   740→   740→  };
